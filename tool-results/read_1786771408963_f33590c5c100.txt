    44→@OptIn(ExperimentalMaterial3Api::class)
    45→@Composable
    46→fun MainScreen(
    47→    viewModel: AgentViewModel,
    48→    onNavigateToSettings: () -> Unit,
    49→    onNavigateToDebug: () -> Unit,
    50→    onNavigateToHistory: () -> Unit
    51→) {
    52→    val state by viewModel.stateFlow.collectAsState()
    53→    val isAccessibilityEnabled by remember {
    54→        mutableStateOf(viewModel.isAccessibilityServiceEnabled())
    55→    }
    56→    var taskInput by remember { mutableStateOf("") }
    57→    val listState = rememberLazyListState()
    58→    val isRunning = state.status.isActive
    59→    val context = LocalContext.current
    60→
    61→    var isListening by remember { mutableStateOf(false) }
    62→
    63→    val displayItems = remember(state.history) {
    64→        state.history.mapNotNull { event ->
    65→            when (event) {
    66→                is AgentEvent.UserMessage -> DisplayItem.UserMessage(event.text, event.timestamp)
    67→                is AgentEvent.AgentMessage -> DisplayItem.AgentMessage(event.text, event.timestamp)
    68→                is AgentEvent.ToolExecution -> DisplayItem.ToolAction(
    69→                    name = event.toolName,
    70→                    args = event.arguments,
    71→                    success = event.result.success,
    72→                    error = event.result.error?.message
    73→                )
    74→                is AgentEvent.Error -> DisplayItem.ErrorItem(event.message)
    75→                else -> null
    76→            }
    77→        }
    78→    }
    79→
    80→    LaunchedEffect(displayItems.size) {
    81→        if (displayItems.isNotEmpty()) {
    82→            listState.animateScrollToItem(displayItems.size - 1)
    83→        }
    84→    }
    85→
    86→    val speechRecognizer = remember {
    87→        try { SpeechRecognizer.createSpeechRecognizer(context) } catch (_: Exception) { null }
    88→    }
    89→    DisposableEffect(Unit) { onDispose { speechRecognizer?.destroy() } }
    90→
    91→    fun startVoiceInput() {
    92→        if (speechRecognizer == null) return
    93→        isListening = true
    94→        val intent = Intent(RecognizerIntent.ACTION_RECOGNIZE_SPEECH).apply {
    95→            putExtra(RecognizerIntent.EXTRA_LANGUAGE_MODEL, RecognizerIntent.LANGUAGE_MODEL_FREE_FORM)
    96→            putExtra(RecognizerIntent.EXTRA_LANGUAGE, "en-IN")
    97→            putExtra(RecognizerIntent.EXTRA_PROMPT, "Speak to Android-Use...")
    98→        }
    99→        try {
   100→            speechRecognizer.setRecognitionListener(object : RecognitionListener {
   101→                override fun onReadyForSpeech(p: Bundle?) {}
   102→                override fun onBeginningOfSpeech() {}
   103→                override fun onRmsChanged(r: Float) {}
   104→                override fun onBufferReceived(b: ByteArray?) {}
   105→                override fun onEndOfSpeech() { isListening = false }
   106→                override fun onError(e: Int) { isListening = false }
   107→                override fun onResults(results: Bundle?) {
   108→                    val matches = results?.getStringArrayList(SpeechRecognizer.RESULTS_RECOGNITION)
   109→                    if (!matches.isNullOrEmpty()) taskInput = matches.first()
   110→                    isListening = false
   111→                }
   112→                override fun onPartialResults(p: Bundle?) {}
   113→                override fun onEvent(e: Int, p: Bundle?) {}
   114→            })
   115→            speechRecognizer.startListening(intent)
   116→        } catch (_: ActivityNotFoundException) { isListening = false }
   117→    }
   118→
   119→    fun sendTask(text: String) {
   120→        val trimmed = text.trim()
   121→        if (trimmed.isNotBlank()) {
   122→            if (state.goal.isBlank()) viewModel.startTask(trimmed)
   123→            else viewModel.continueChat(trimmed)
   124→            taskInput = ""
   125→        }
   126→    }
   127→
   128→    Box(modifier = Modifier.fillMaxSize().background(AppColors.DarkBackground)) {
   129→        Column(modifier = Modifier.fillMaxSize().align(Alignment.TopStart)) {
   130→            TopBar(
   131→                onNewChat = { viewModel.newChat() },
   132→                onHistory = onNavigateToHistory,
   133→                onSettings = onNavigateToSettings,
   134→                onDebug = onNavigateToDebug
   135→            )
   136→            if (isRunning) RunningStatusBar(state = state)
   137→            if (!isAccessibilityEnabled) {
   138→                AccessibilityBanner(onEnable = { viewModel.openAccessibilitySettings() })
   139→            }
   140→            if (state.status == AgentStatus.COMPLETED && !isRunning) {
   141→                ResultBanner(text = "Done", isSuccess = true)
   142→            } else if (state.status == AgentStatus.FAILED && !isRunning) {
   143→                ResultBanner(text = state.lastError ?: "Something went wrong", isSuccess = false)
   144→            } else if (state.status == AgentStatus.CANCELLED && !isRunning) {
   145→                ResultBanner(text = "Stopped", isSuccess = false)
   146→            }
   147→
   148→            if (displayItems.isEmpty() && !isRunning) {
   149→                EmptyState(onSuggestionClick = { sendTask(it) })
   150→            } else {
   151→                LazyColumn(
   152→                    state = listState,
   153→                    modifier = Modifier.fillMaxSize().padding(horizontal = 14.dp),
   154→                    verticalArrangement = Arrangement.spacedBy(5.dp),
   155→                    contentPadding = PaddingValues(vertical = 6.dp)
   156→                ) {
   157→                    items(items = displayItems, key = { it.timestamp.toString() + it::class.simpleName }) { item ->
   158→                        when (item) {
   159→                            is DisplayItem.UserMessage -> UserChatBubble(text = item.text)
   160→                            is DisplayItem.AgentMessage -> AgentChatBubble(text = item.text)
   161→                            is DisplayItem.ToolAction -> ToolActionCard(item)
   162→                            is DisplayItem.ErrorItem -> ErrorText(item.message)
   163→                        }
   164→                    }
   165→                    item { Spacer(modifier = Modifier.height(90.dp)) }
   166→                }
   167→            }
   168→        }
   169→
   170→        // ── BOTTOM INPUT BAR — anchored to bottom ──
   171→        Box(
   172→            modifier = Modifier
   173→                .align(Alignment.BottomCenter)
   174→                .fillMaxWidth()
   175→                .background(AppColors.DarkBackground)
   176→        ) {
   177→            BottomInputBar(
   178→                input = taskInput,
   179→                onInputChange = { taskInput = it },
   180→                onSend = { sendTask(taskInput) },
   181→                onStop = { viewModel.stopAgent() },
   182→                isRunning = isRunning,
   183→                isListening = isListening,
   184→                onVoiceClick = { startVoiceInput() },
   185→                canSend = taskInput.isNotBlank() && !isRunning
   186→            )
   187→        }
   188→
   189→        state.pendingConfirmation?.let { conf ->
   190→            ConfirmationDialog(conf, onConfirm = { viewModel.respondToConfirmation(true) }, onDeny = { viewModel.respondToConfirmation(false) })
   191→        }
   192→        state.pendingQuestion?.let { q ->
   193→            UserQuestionDialog(q, onAnswer = { viewModel.respondToUser(it) }, onDismiss = {})
   194→        }
   195→    }
   196→}

   258→@Composable
   259→private fun RunningStatusBar(state: com.androidagent.aiagent.agent.AgentState) {
   260→    val (label, icon, color) = when (state.status) {
   261→        AgentStatus.THINKING -> Triple("Analyzing screen", Icons.Default.Psychology, AppColors.TextSecondary)
   262→        AgentStatus.EXECUTING -> Triple("Using mobile...", Icons.Default.TouchApp, AppColors.Success)
   263→        AgentStatus.WAITING_FOR_USER -> Triple("Waiting for you", Icons.Default.QuestionAnswer, AppColors.Primary)
   264→        AgentStatus.WAITING_FOR_CONFIRMATION -> Triple("Confirm action", Icons.Default.Security, AppColors.Warning)
   265→        AgentStatus.VERIFYING -> Triple("Verifying...", Icons.Default.VerifiedUser, AppColors.TextSecondary)
   266→        else -> Triple("Working...", Icons.Default.Autorenew, AppColors.Secondary)
   267→    }
   268→    val pulse = rememberInfiniteTransition(label = "p").animateFloat(
   269→        0.3f, 1f, infiniteRepeatable(tween(800, LinearEasing), RepeatMode.Reverse), "a"
   270→    )
   271→    Row(
   272→        modifier = Modifier.fillMaxWidth().background(color.copy(alpha = 0.06f)).padding(horizontal = 14.dp, vertical = 6.dp),
   273→        verticalAlignment = Alignment.CenterVertically
   274→    ) {
   275→        Icon(icon, null, tint = color, modifier = Modifier.size(14.dp).alpha(pulse.value))
   276→        Spacer(Modifier.width(8.dp))
   277→        Text(label, color = color, fontSize = 12.5.sp, fontWeight = FontWeight.Medium, modifier = Modifier.weight(1f))
   278→        Text("${state.durationFormatted}", color = AppColors.TextMuted, fontSize = 11.sp)
   279→        Spacer(Modifier.width(10.dp))
   280→        Text("Step ${state.stepNumber}", color = AppColors.TextMuted, fontSize = 11.sp)
   281→    }
   282→}

   356→private fun BottomInputBar(
   357→    input: String, onInputChange: (String) -> Unit, onSend: () -> Unit,
   358→    onStop: () -> Unit, isRunning: Boolean, isListening: Boolean,
   359→    onVoiceClick: () -> Unit, canSend: Boolean
   360→) {

   362→        modifier = Modifier.fillMaxWidth().background(AppColors.Surface).navigationBarsPadding().padding(horizontal = 12.dp, vertical = 8.dp)
   363→    ) {

   365→            val s = rememberInfiniteTransition(label = "m").animateFloat(0.8f, 1.2f, infiniteRepeatable(tween(400), RepeatMode.Reverse), "s")
   366→            Row(modifier = Modifier.fillMaxWidth().padding(bottom = 5.dp), horizontalArrangement = Arrangement.Center) {
   367→                Icon(Icons.Default.Mic, "Listening", tint = AppColors.Error, modifier = Modifier.size(20.dp * s.value))

   370→            }
   371→        }
   372→        Surface(
   373→            color = AppColors.SurfaceVariant, shape = RoundedCornerShape(24.dp),
   374→            border = BorderStroke(1.dp, AppColors.Line), modifier = Modifier.fillMaxWidth()
   375→        ) {
   376→            Row(modifier = Modifier.padding(horizontal = 4.dp, vertical = 2.dp), verticalAlignment = Alignment.CenterVertically) {
   377→                IconButton(onClick = onVoiceClick, modifier = Modifier.size(42.dp)) {
   378→                    Icon(
   379→                        if (isListening) Icons.Default.Mic else Icons.Default.MicNone,
   380→                        "Voice", tint = if (isListening) AppColors.Error else AppColors.TextMuted, modifier = Modifier.size(20.dp)
   381→                    )
   382→                }
   383→                BasicTextField(
   384→                    value = input, onValueChange = onInputChange, modifier = Modifier.weight(1f),
   385→                    textStyle = TextStyle(color = AppColors.TextPrimary, fontSize = 15.sp, lineHeight = 20.sp),
   386→                    cursorBrush = SolidColor(AppColors.TextPrimary),
   387→                    decorationBox = { inner ->
   388→                        Box(Modifier.padding(vertical = 10.dp)) {
   389→                            if (input.isEmpty()) Text("Ask Android-Use anything...", color = AppColors.TextMuted, fontSize = 15.sp)
   390→                            inner()
   391→                        }
   392→                    }
   393→                )
   394→                Box(
   395→                    modifier = Modifier.size(38.dp).clip(CircleShape).background(
   396→                        when { isRunning -> AppColors.Error; canSend -> AppColors.TextPrimary; else -> AppColors.SurfaceHover }
   397→                    ).clickable(enabled = canSend || isRunning) {
   398→                        if (isRunning) onStop() else onSend()
   399→                    }, contentAlignment = Alignment.Center
   400→                ) {
   401→                    Icon(
   402→                        if (isRunning) Icons.Default.Stop else Icons.AutoMirrored.Filled.Send,
   403→                        if (isRunning) "Stop" else "Send",
   404→                        tint = when { isRunning -> Color.White; canSend -> AppColors.DarkBackground; else -> AppColors.TextMuted },
   405→                        modifier = Modifier.size(18.dp)
   406→                    )
   407→                }
   408→            }
   409→        }
   410→    }
   411→}

   576→private fun extractJsonField(json: String, field: String): String? {
   577→    return Regex("""$field"\s*:\s*"([^"\x27]*)"""", RegexOption.IGNORE_CASE).find(json)?.groupValues?.getOrNull(1)
