   171→@OptIn(ExperimentalMaterial3Api::class)
   172→@Composable
   173→fun EventCard(
   174→    event: AgentEvent,
   175→    expanded: Boolean,
   176→    onToggle: () -> Unit
   177→) {
   178→    val (icon, iconColor, title) = when (event) {
   179→        is AgentEvent.ToolExecution -> {
   180→            val isSuccess = event.result.success
   181→            Triple(
   182→                if (isSuccess) Icons.Default.CheckCircle else Icons.Default.Error,
   183→                if (isSuccess) AppColors.Success else AppColors.Error,
   184→                "Tool: ${event.toolName}"
   185→            )
   186→        }
   187→        is AgentEvent.Observation -> Triple(
   188→            Icons.Default.KeyboardArrowDown,
   189→            AppColors.Primary,
   190→            "Observation"
   191→        )
   192→        is AgentEvent.ModelResponse -> Triple(
   193→            Icons.AutoMirrored.Filled.ArrowRight,
   194→            AppColors.Primary,
   195→            "Model: ${event.decisionType}"
   196→        )
   197→        is AgentEvent.UserMessage -> Triple(
   198→            Icons.Default.KeyboardArrowDown,
   199→            AppColors.Primary,
   200→            "User: ${event.text.take(50)}"
   201→        )
   202→        is AgentEvent.StatusChange -> Triple(
   203→            Icons.Default.KeyboardArrowDown,
   204→            AppColors.Secondary,
   205→            "${event.from} → ${event.to}"
   206→        )
   207→        is AgentEvent.Error -> Triple(
   208→            Icons.Default.Error,
   209→            AppColors.Error,
   210→            event.message.take(80)
   211→        )
   212→    }
   213→
   214→    Card(
   215→        modifier = Modifier
   216→            .fillMaxWidth()
   217→            .padding(horizontal = 16.dp, vertical = 2.dp)
   218→            .clickable { onToggle() },
   219→        colors = CardDefaults.cardColors(
   220→            containerColor = AppColors.Surface
   221→        ),
   222→        shape = RoundedCornerShape(12.dp)
   223→    ) {
   224→        Column(
   225→            modifier = Modifier.padding(12.dp)
   226→        ) {
   227→            Row(
   228→                modifier = Modifier.fillMaxWidth(),
   229→                verticalAlignment = Alignment.CenterVertically
   230→            ) {
   231→                Icon(
   232→                    imageVector = icon,
   233→                    contentDescription = null,
   234→                    tint = iconColor,
   235→                    modifier = Modifier.size(18.dp)
   236→                )
   237→                Spacer(modifier = Modifier.width(8.dp))
   238→                Text(
   239→                    text = title,
   240→                    style = MaterialTheme.typography.bodyMedium,
   241→                    color = AppColors.TextPrimary,
   242→                    maxLines = 1,
   243→                    overflow = TextOverflow.Ellipsis,
   244→                    modifier = Modifier.weight(1f)
   245→                )
   246→                Text(
   247→                    text = "Step ${event.stepNumber}",
   248→                    style = MaterialTheme.typography.labelSmall,
   249→                    color = AppColors.TextMuted
   250→                )
   251→                Spacer(modifier = Modifier.width(4.dp))
   252→                Icon(
   253→                    imageVector = if (expanded) Icons.Default.ExpandLess else Icons.Default.ExpandMore,
   254→                    contentDescription = if (expanded) "Collapse" else "Expand",
   255→                    tint = AppColors.TextSecondary,
   256→                    modifier = Modifier.size(18.dp)
   257→                )
   258→            }
   259→
   260→            AnimatedVisibility(visible = expanded) {
   261→                Column(
   262→                    modifier = Modifier.padding(top = 8.dp)
   263→                ) {
   264→                    when (event) {
   265→                        is AgentEvent.ToolExecution -> ToolExecutionDetail(event)
   266→                        is AgentEvent.Observation -> {
   267→                            Text(
   268→                                text = event.summary,
   269→                                style = MaterialTheme.typography.bodySmall,
   270→                                color = AppColors.TextSecondary,
   271→                                fontFamily = FontFamily.Monospace
   272→                            )
   273→                        }
   274→                        is AgentEvent.ModelResponse -> {
   275→                            Text(
   276→                                text = event.content,
   277→                                style = MaterialTheme.typography.bodySmall,
   278→                                color = AppColors.TextSecondary
   279→                            )
   280→                        }
   281→                        is AgentEvent.UserMessage -> {
   282→                            Text(
   283→                                text = event.text,
   284→                                style = MaterialTheme.typography.bodySmall,
   285→                                color = AppColors.TextSecondary
   286→                            )
   287→                        }
   288→                        is AgentEvent.StatusChange -> {
   289→                            Text(
   290→                                text = "${event.from} → ${event.to}",
   291→                                style = MaterialTheme.typography.bodySmall,
   292→                                color = AppColors.TextSecondary
   293→                            )
   294→                        }
   295→                        is AgentEvent.Error -> {
   296→                            Text(
   297→                                text = event.message,
   298→                                style = MaterialTheme.typography.bodySmall,
   299→                                color = AppColors.Error
   300→                            )
   301→                        }
   302→                    }
   303→                }
   304→            }
   305→        }
   306→    }
   307→}
