  1770→  1770→/* If a previous turn was interrupted (crash/reload/close) mid-task, offer to
  1771→  1771→   resume it so nothing is silently lost. */
  1772→  1772→function checkResume(){
  1773→  1773→  let r;try{r=JSON.parse(localStorage.getItem(LS_RESUME)||"null");}catch(e){}
  1774→  1774→  if(!r||!r.chatId)return;
  1775→  1775→  const chat=chats.find(c=>c.id===r.chatId);
  1776→  1776→  if(!chat||!chat.messages.length){clearResume();return;}
  1777→  1777→  const last=chat.messages[chat.messages.length-1];
  1778→  1778→  // Only offer if the last turn wasn't a completed assistant answer.
  1779→  1779→  if(last.role==="assistant"&&last.text&&(!last.tool_calls||!last.tool_calls.length)){clearResume();return;}
  1780→  1780→  let lastUser="";for(let i=chat.messages.length-1;i>=0;i--){if(chat.messages[i].role==="user"){lastUser=chat.messages[i].text;break;}}
  1781→  1781→  if(!lastUser){clearResume();return;}
  1782→  1782→  const t=document.createElement("div");t.className="toast";t.style.pointerEvents="auto";
  1783→  1783→  t.innerHTML=`<svg class="ic ti"><use href="#i-refresh"/></svg>Task was interrupted. <button class="btn ghost" style="padding:4px 10px;font-size:12px;margin-left:6px">Resume</button>`;
  1784→  1784→  t.querySelector("button").onclick=()=>{t.remove();openChat(r.chatId);
  1785→  1785→    // trim any dangling partial assistant/tool turn, then re-run the last user msg
  1786→  1786→    for(let i=chat.messages.length-1;i>=0;i--){if(chat.messages[i].role==="user"){chat.messages=chat.messages.slice(0,i);break;}}
  1787→  1787→    chat.used=0;chat.cost=0;saveChats();openChat(r.chatId);agentTurn(lastUser);};
  1788→  1788→  $("#toasts").appendChild(t);clearResume();
  1789→  1789→}
