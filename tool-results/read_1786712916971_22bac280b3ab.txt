    43→    43→@OptIn(ExperimentalMaterial3Api::class)
    44→    44→@Composable
    45→    45→fun MainScreen(
    46→    46→    viewModel: AgentViewModel,
    47→    47→    onNavigateToSettings: () -> Unit,
    48→    48→    onNavigateToDebug: () -> Unit,
    49→    49→    onNavigateToHistory: () -> Unit
    50→    50→) {
    51→    51→    val state by viewModel.stateFlow.collectAsState()
    52→    52→    val isAccessibilityEnabled by remember {
    53→    53→        mutableStateOf(viewModel.isAccessibilityServiceEnabled())
    54→    54→    }
    55→    55→    var taskInput by remember { mutableStateOf("") }
    56→    56→    val listState = rememberLazyListState()
    57→    57→    val isRunning = state.status.isActive
    58→    58→    val context = LocalContext.current
    59→    59→
    60→    60→    // Voice input state
    61→    61→    var isListening by remember { mutableStateOf(false) }
    62→    62→    var voiceText by remember { mutableStateOf("") }
    63→    63→
    64→    64→    // Build display items: chat messages + tool actions
    65→    65→    val displayItems = remember(state.history) {
    66→    66→        state.history.mapNotNull { event ->
    67→    67→            when (event) {
    68→    68→                is AgentEvent.UserMessage -> DisplayItem.UserMessage(event.text, event.timestamp)
    69→    69→                is AgentEvent.AgentMessage -> DisplayItem.AgentMessage(event.text, event.timestamp)
    70→    70→                is AgentEvent.ToolExecution -> DisplayItem.ToolAction(
    71→    71→                    name = event.toolName,
    72→    72→                    args = event.arguments,
    73→    73→                    success = event.result.success,
    74→    74→                    error = event.result.error?.message
    75→    75→                )
    76→    76→                is AgentEvent.Error -> DisplayItem.ErrorItem(event.message)
    77→    77→                else -> null
    78→    78→            }
    79→    79→        }
    80→    80→    }
    81→    81→
    82→    82→    LaunchedEffect(displayItems.size) {
    83→    83→        if (displayItems.isNotEmpty()) {
    84→    84→            listState.animateScrollToItem(displayItems.size - 1)
    85→    85→        }
    86→    86→    }
    87→    87→
    88→    88→    // Voice recognizer
    89→    89→    val speechRecognizer = remember {
    90→    90→        try {
    91→    91→            SpeechRecognizer.createSpeechRecognizer(context)
    92→    92→        } catch (e: Exception) { null }
    93→    93→    }
    94→    94→
    95→    95→    DisposableEffect(Unit) {
    96→    96→        onDispose {
    97→    97→            speechRecognizer?.destroy()
    98→    98→        }
    99→    99→    }
   100→   100→
   101→   101→    fun startVoiceInput() {
   102→   102→        if (speechRecognizer == null) return
   103→   103→        isListening = true
   104→   104→        voiceText = ""
   105→   105→        val intent = Intent(RecognizerIntent.ACTION_RECOGNIZE_SPEECH).apply {
   106→   106→            putExtra(RecognizerIntent.EXTRA_LANGUAGE_MODEL, RecognizerIntent.LANGUAGE_MODEL_FREE_FORM)
   107→   107→            putExtra(RecognizerIntent.EXTRA_LANGUAGE, "en-IN")
   108→   108→            putExtra(RecognizerIntent.EXTRA_PROMPT, "Speak to Android-Use...")
   109→   109→        }
   110→   110→        try {
   111→   111→            speechRecognizer.setRecognitionListener(object : RecognitionListener {
   112→   112→                override fun onReadyForSpeech(params: Bundle?) {}
   113→   113→                override fun onBeginningOfSpeech() {}
   114→   114→                override fun onRmsChanged(rmsdB: Float) {}
   115→   115→                override fun onBufferReceived(buffer: ByteArray?) {}
   116→   116→                override fun onEndOfSpeech() {
   117→   117→                    isListening = false
   118→   118→                }
   119→   119→                override fun onError(error: Int) {
   120→   120→                    isListening = false
   121→   121→                }
   122→   122→                override fun onResults(results: Bundle?) {
   123→   123→                    val matches = results?.getStringArrayList(SpeechRecognizer.RESULTS_RECOGNITION)
   124→   124→                    val text = matches?.firstOrNull() ?: ""
   125→   125→                    if (text.isNotBlank()) {
   126→   126→                        taskInput = text
   127→   127→                    }
   128→   128→                    isListening = false
   129→   129→                }
   130→   130→                override fun onPartialResults(partialResults: Bundle?) {
   131→   131→                    val matches = partialResults?.getStringArrayList(SpeechRecognizer.RESULTS_RECOGNITION)
   132→   132→                    voiceText = matches?.firstOrNull() ?: ""
   133→   133→                }
   134→   134→                override fun onEvent(eventType: Int, params: Bundle?) {}
   135→   135→            })
   136→   136→            speechRecognizer.startListening(intent)
   137→   137→        } catch (e: ActivityNotFoundException) {
   138→   138→            isListening = false
   139→   139→        }
   140→   140→    }
   141→   141→
   142→   142→    Box(
   143→   143→        modifier = Modifier
   144→   144→            .fillMaxSize()
   145→   145→            .background(AppColors.DarkBackground)
   146→   146→    ) {
   147→   147→        Column(modifier = Modifier.fillMaxSize()) {
   148→   148→            // ── Top Bar ──
   149→   149→            TopBar(
   150→   150→                onNewChat = { viewModel.newChat() },
   151→   151→                onHistory = onNavigateToHistory,
   152→   152→                onSettings = onNavigateToSettings
   153→   153→            )
   154→   154→
   155→   155→            // ── Status bar (when running) ──
   156→   156→            if (isRunning) {
   157→   157→                RunningStatusBar(state = state)
   158→   158→            }
   159→   159→
   160→   160→            // ── Accessibility banner ──
   161→   161→            if (!isAccessibilityEnabled) {
   162→   162→                AccessibilityBanner(onEnable = { viewModel.openAccessibilitySettings() })
   163→   163→            }
   164→   164→
   165→   165→            // ── Result banners ──
   166→   166→            if (state.status == AgentStatus.COMPLETED && !isRunning) {
   167→   167→                ResultBanner(text = "Done", isSuccess = true)
   168→   168→            } else if (state.status == AgentStatus.FAILED && !isRunning) {
   169→   169→                ResultBanner(text = state.lastError ?: "Something went wrong", isSuccess = false)
   170→   170→            } else if (state.status == AgentStatus.CANCELLED && !isRunning) {
   171→   171→                ResultBanner(text = "Stopped", isSuccess = false)
   172→   172→            }
   173→   173→
   174→   174→            // ── Chat / Event List ──
   175→   175→            if (displayItems.isEmpty() && !isRunning) {
   176→   176→                EmptyState()
   177→   177→            } else {
   178→   178→                LazyColumn(
   179→   179→                    state = listState,
   180→   180→                    modifier = Modifier
   181→   181→                        .fillMaxSize()
   182→   182→                        .padding(horizontal = 16.dp),
   183→   183→                    verticalArrangement = Arrangement.spacedBy(6.dp),
   184→   184→                    contentPadding = PaddingValues(vertical = 8.dp)
   185→   185→                ) {
   186→   186→                    items(
   187→   187→                        items = displayItems,
   188→   188→                        key = { it.timestamp.toString() + it::class.simpleName }
   189→   189→                    ) { item ->
   190→   190→                        when (item) {
   191→   191→                            is DisplayItem.UserMessage -> UserChatBubble(text = item.text)
   192→   192→                            is DisplayItem.AgentMessage -> AgentChatBubble(text = item.text)
   193→   193→                            is DisplayItem.ToolAction -> ToolActionCard(item)
   194→   194→                            is DisplayItem.ErrorItem -> ErrorText(item.message)
   195→   195→                        }
   196→   196→                    }
   197→   197→                    // Bottom padding for input bar
   198→   198→                    item {
   199→   199→                        Spacer(modifier = Modifier.height(80.dp))
   200→   200→                    }
   201→   201→                }
   202→   202→            }
   203→   203→        }
   204→   204→
   205→   205→        // ── Bottom Input Bar ──
   206→   206→        BottomInputBar(
   207→   207→            input = taskInput,
   208→   208→            onInputChange = { taskInput = it },
   209→   209→            onSend = {
   210→   210→                val trimmed = taskInput.trim()
   211→   211→                if (trimmed.isNotBlank()) {
   212→   212→                    if (state.goal.isBlank()) {
   213→   213→                        viewModel.startTask(trimmed)
   214→   214→                    } else {
   215→   215→                        viewModel.continueChat(trimmed)
   216→   216→                    }
   217→   217→                    taskInput = ""
   218→   218→                }
   219→   219→            },
   220→   220→            onStop = { viewModel.stopAgent() },
   221→   221→            isRunning = isRunning,
   222→   222→            isListening = isListening,
   223→   223→            onVoiceClick = { startVoiceInput() },
   224→   224→            canSend = taskInput.isNotBlank() && !isRunning
   225→   225→        )
   226→   226→
   227→   227→        // ── Dialogs ──
   228→   228→        state.pendingConfirmation?.let { confirmation ->
   229→   229→            ConfirmationDialog(
   230→   230→                confirmation = confirmation,
   231→   231→                onConfirm = { viewModel.respondToConfirmation(true) },
   232→   232→                onDeny = { viewModel.respondToConfirmation(false) }
   233→   233→            )
   234→   234→        }
   235→   235→
   236→   236→        state.pendingQuestion?.let { question ->
   237→   237→            UserQuestionDialog(
   238→   238→                question = question,
   239→   239→                onAnswer = { answer -> viewModel.respondToUser(answer) },
   240→   240→                onDismiss = {}
   241→   241→            )
   242→   242→        }
   243→   243→    }
   244→   244→}

   463→   463→private fun BottomInputBar(
   464→   464→    input: String,
   465→   465→    onInputChange: (String) -> Unit,
   466→   466→    onSend: () -> Unit,
   467→   467→    onStop: () -> Unit,
   468→   468→    isRunning: Boolean,

   477→   477→            .navigationBarsPadding()
   478→   478→            .padding(horizontal = 12.dp, vertical = 8.dp)
   479→   479→    ) {
   480→   480→        // Listening indicator

   699→   699→private fun describeAction(toolName: String, arguments: String): String {
   700→   700→    val text = extractJsonField(arguments, "text")
   701→   701→    val appName = extractJsonField(arguments, "app_name")
   702→   702→    val direction = extractJsonField(arguments, "direction")
   703→   703→    return when {
   704→   704→        toolName.contains("launch_app") -> if (!appName.isNullOrBlank()) "Opened $appName" else "Opened app"
   705→   705→        toolName.contains("click") -> "Tapped element"
   706→   706→        toolName.contains("long_click") -> "Long pressed"
   707→   707→        toolName.contains("type_text") -> if (!text.isNullOrBlank()) "Typed: ${text.take(40)}" else "Typed text"
   708→   708→        toolName.contains("clear_text") -> "Cleared text"
   709→   709→        toolName.contains("scroll") -> "Scrolled ${direction ?: ""}"
   710→   710→        toolName.contains("swipe") -> {
   711→   711→            val d = extractJsonField(arguments, "direction") ?: ""
   712→   712→            when {
   713→   713→                d.contains("up", ignoreCase = true) -> "Swiped up"
   714→   714→                d.contains("down", ignoreCase = true) -> "Swiped down"
   715→   715→                d.contains("left", ignoreCase = true) -> "Swiped left"
   716→   716→                d.contains("right", ignoreCase = true) -> "Swiped right"
   717→   717→                else -> "Swiped"
   718→   718→            }
   719→   719→        }
   720→   720→        toolName.contains("back") -> "Went back"
   721→   721→        toolName.contains("home") -> "Went home"
   722→   722→        toolName.contains("recents") -> "Opened recents"
   723→   723→        toolName.contains("press_key") -> "Pressed key"
   724→   724→        toolName.contains("wait") -> "Waiting..."
   725→   725→        toolName.contains("screenshot") -> "Took screenshot"
   726→   726→        toolName.contains("inspect") -> "Read screen"
   727→   727→        toolName.contains("find") -> "Searching..."
   728→   728→        toolName.contains("analyze") -> "Analyzing screen"
   729→   729→        toolName.contains("visual") -> "Looking for element"
   730→   730→        toolName.contains("finish") -> "Done"
   731→   731→        toolName.contains("stop") -> "Stopped"
   732→   732→        else -> toolName.substringAfterLast(".").replace("_", " ")
   733→   733→    }
   734→   734→}

   753→   753→private fun extractJsonField(json: String, field: String): String? {
