   176→@OptIn(ExperimentalMaterial3Api::class)
   177→@Composable
   178→fun EventCard(
   179→    event: AgentEvent,
   180→    expanded: Boolean,
   181→    onToggle: () -> Unit
   182→) {
   183→    val (icon, iconColor, title) = when (event) {
   184→        is AgentEvent.ToolExecution -> {
   185→            val isSuccess = event.result.contains("success", ignoreCase = true) ||
   186→                    !event.result.contains("error", ignoreCase = true)
   187→            Triple(
   188→                if (isSuccess) Icons.Default.CheckCircle else Icons.Default.Error,
   189→                if (isSuccess) AppColors.Success else AppColors.Error,
   190→                "Tool: ${event.toolName}"
   191→            )
   192→        }
   193→        is AgentEvent.Observation -> Triple(
   194→            Icons.Default.KeyboardArrowDown,
   195→            AppColors.Primary,
   196→            "Observation"
   197→        )
   198→        is AgentEvent.ModelResponse -> Triple(
   199→            Icons.AutoMirrored.Filled.ArrowRight,
   200→            AppColors.Primary,
   201→                "Model: ${event.decisionType}"
   202→            )
   203→        is AgentEvent.UserMessage -> Triple(
   204→            Icons.Default.KeyboardArrowDown,
   205→            AppColors.Primary,
   206→            "User: ${event.text.take(50)}"
   207→        )
   208→        is AgentEvent.StatusChange -> Triple(
   209→            Icons.Default.KeyboardArrowDown,
   210→            AppColors.Secondary,
   211→            "${event.from} → ${event.to}"
   212→        )
   213→        is AgentEvent.Error -> Triple(
   214→            Icons.Default.Error,
   215→            AppColors.Error,
   216→            event.message.take(80)
   217→        )
   218→    }
   219→
   220→    Card(
   221→        modifier = Modifier
   222→            .fillMaxWidth()
   223→            .padding(horizontal = 16.dp, vertical = 2.dp)
   224→            .clickable { onToggle() },
   225→        colors = CardDefaults.cardColors(
   226→            containerColor = AppColors.Surface
   227→        ),
   228→        shape = RoundedCornerShape(12.dp)
   229→    ) {
   230→        Column(
   231→            modifier = Modifier.padding(12.dp)
   232→        ) {
   233→            Row(
   234→                modifier = Modifier.fillMaxWidth(),
   235→                verticalAlignment = Alignment.CenterVertically
   236→            ) {
   237→                Icon(
   238→                    imageVector = icon,
   239→                    contentDescription = null,
   240→                    tint = iconColor,
   241→                    modifier = Modifier.size(18.dp)
   242→                )
   243→                Spacer(modifier = Modifier.width(8.dp))
   244→                Text(
   245→                    text = title,
   246→                    style = MaterialTheme.typography.bodyMedium,
   247→                    color = AppColors.TextPrimary,
   248→                    maxLines = 1,
   249→                    overflow = TextOverflow.Ellipsis,
   250→                    modifier = Modifier.weight(1f)
   251→                )
   252→                Text(
   253→                    text = "Step ${event.stepNumber}",
   254→                    style = MaterialTheme.typography.labelSmall,
   255→                    color = AppColors.TextMuted
   256→                )
   257→                Spacer(modifier = Modifier.width(4.dp))
   258→                Icon(
   259→                    imageVector = if (expanded) Icons.Default.ExpandLess else Icons.Default.ExpandMore,
   260→                    contentDescription = if (expanded) "Collapse" else "Expand",
   261→                    tint = AppColors.TextSecondary,
   262→                    modifier = Modifier.size(18.dp)
   263→                )
   264→            }
   265→
   266→            AnimatedVisibility(visible = expanded) {
   267→                Column(
   268→                    modifier = Modifier.padding(top = 8.dp)
   269→                ) {
   270→                    when (event) {
   271→                        is AgentEvent.ToolExecution -> ToolExecutionDetail(event)
   272→                        is AgentEvent.Observation -> {
   273→                            Text(
   274→                                text = event.summary,
   275→                                style = MaterialTheme.typography.bodySmall,
   276→                                color = AppColors.TextSecondary,
   277→                                fontFamily = FontFamily.Monospace
   278→                            )
   279→                        }
   280→                        is AgentEvent.ModelResponse -> {
   281→                            Text(
   282→                                text = event.content,
   283→                                style = MaterialTheme.typography.bodySmall,
   284→                                color = AppColors.TextSecondary
   285→                            )
   286→                        }
   287→                        is AgentEvent.UserMessage -> {
   288→                            Text(
   289→                                text = event.text,
   290→                                style = MaterialTheme.typography.bodySmall,
   291→                                color = AppColors.TextSecondary
   292→                            )
   293→                        }
   294→                        is AgentEvent.StatusChange -> {
   295→                            Text(
   296→                                text = "${event.from} → ${event.to}",
   297→                                style = MaterialTheme.typography.bodySmall,
   298→                                color = AppColors.TextSecondary
   299→                            )
   300→                        }
   301→                        is AgentEvent.Error -> {
   302→                            Text(
   303→                                text = event.message,
   304→                                style = MaterialTheme.typography.bodySmall,
   305→                                color = AppColors.Error
   306→                            )
   307→                        }
   308→                    }
   309→                }
   310→            }
   311→        }
   312→    }
   313→}

   597→        shape = RoundedCornerShape(16.dp)
