   164→// Step Cards (colored left border, smooth expand/collapse)

   166→
   167→@Composable
   168→fun EventCard(
   169→    event: AgentEvent,
   170→    expanded: Boolean,
   171→    onToggle: () -> Unit
   172→) {
   173→    data class EventInfo(val icon: ImageVector, val accentColor: Color, val title: String, val stepLabel: String)
   174→
   175→    val info = when (event) {
   176→        is AgentEvent.ToolExecution -> {
   177→            val isSuccess = event.result.success
   178→            EventInfo(
   179→                icon = if (isSuccess) Icons.Default.CheckCircle else Icons.Default.Error,
   180→                accentColor = if (isSuccess) AppColors.SuccessGreen else AppColors.ErrorRed,
   181→                title = event.toolName,
   182→                stepLabel = "Tool"
   183→            )
   184→        }
   185→        is AgentEvent.Observation -> EventInfo(
   186→            Icons.Default.KeyboardArrowDown,
   187→            AppColors.ObservationPurple,
   188→            "Screen captured",
   189→            "Observation"
   190→        )
   191→        is AgentEvent.ModelResponse -> EventInfo(
   192→            Icons.AutoMirrored.Filled.ArrowRight,
   193→            AppColors.AccentBlue,
   194→            event.decisionType,
   195→            "Model"
   196→        )
   197→        is AgentEvent.UserMessage -> EventInfo(
   198→            Icons.Default.KeyboardArrowDown,
   199→            AppColors.AccentBlue,
   200→            event.text.take(60),
   201→            "User"
   202→        )
   203→        is AgentEvent.StatusChange -> EventInfo(
   204→            Icons.Default.KeyboardArrowDown,
   205→            AppColors.TextSecondary,
   206→            "${event.from} \u2192 ${event.to}",
   207→            "Status"
   208→        )
   209→        is AgentEvent.Error -> EventInfo(
   210→            Icons.Default.Error,
   211→            AppColors.ErrorRed,
   212→            event.message.take(80),
   213→            "Error"
   214→        )
   215→    }
   216→
   217→    val (icon, accentColor, title, stepLabel) = info
   218→
   219→    // Chevron rotation animation
   220→    val rotation by animateFloatAsState(
   221→        targetValue = if (expanded) 180f else 0f,
   222→        animationSpec = androidx.compose.animation.core.tween(250)
   223→    )
   224→
   225→    Column(
   226→        modifier = Modifier
   227→            .fillMaxWidth()
   228→            .padding(horizontal = 16.dp, vertical = 2.dp)
   229→            .clip(RoundedCornerShape(10.dp))
   230→            .background(AppColors.Surface)
   231→            .border(1.dp, AppColors.SurfaceBorder, RoundedCornerShape(10.dp))
   232→            .clickable(onClick = onToggle)
   233→    ) {
   234→        // Header row with colored left accent bar
   235→        Row(
   236→            modifier = Modifier
   237→                .fillMaxWidth()
   238→                .padding(start = 12.dp, end = 8.dp, top = 10.dp, bottom = 10.dp),
   239→            verticalAlignment = Alignment.CenterVertically
   240→        ) {
   241→            // Colored left accent bar
   242→            Box(
   243→                modifier = Modifier
   244→                    .width(3.dp)
   245→                    .height(24.dp)
   246→                    .clip(RoundedCornerShape(2.dp))
   247→                    .background(accentColor)
   248→            )
   249→
   250→            Spacer(modifier = Modifier.width(10.dp))
   251→
   252→            // Step type label (small, muted)
   253→            Text(
   254→                text = stepLabel,
   255→                style = MaterialTheme.typography.labelSmall,
   256→                color = AppColors.TextMuted,
   257→                fontWeight = FontWeight.Medium,
   258→                letterSpacing = 1.sp
   259→            )
   260→
   261→            Spacer(modifier = Modifier.width(6.dp))
   262→
   263→            // Title
   264→            Text(
   265→                text = title,
   266→                style = MaterialTheme.typography.bodyMedium,
   267→                color = AppColors.TextPrimary,
   268→                maxLines = 1,
   269→                overflow = TextOverflow.Ellipsis,
   270→                fontWeight = FontWeight.Medium,
   271→                modifier = Modifier.weight(1f)
   272→            )
   273→
   274→            // Step number
   275→            Text(
   276→                text = "#${event.stepNumber}",
   277→                style = MaterialTheme.typography.labelSmall,
   278→                color = AppColors.TextMuted,
   279→                fontFamily = FontFamily.Monospace
   280→            )
   281→
   282→            Spacer(modifier = Modifier.width(4.dp))
   283→
   284→            // Expand chevron with rotation
   285→            Icon(
   286→                imageVector = Icons.Default.ExpandMore,
   287→                contentDescription = if (expanded) "Collapse" else "Expand",
   288→                tint = AppColors.TextSecondary,
   289→                modifier = Modifier
   290→                    .size(18.dp)
   291→                    .rotate(rotation)
   292→            )
   293→        }
   294→
   295→        // Animated detail section
   296→        AnimatedVisibility(
   297→            visible = expanded,
   298→            enter = expandVertically() + fadeIn(),
   299→            exit = shrinkVertically() + fadeOut()
   300→        ) {
   301→            Column(
   302→                modifier = Modifier
   303→                    .fillMaxWidth()
   304→                    .padding(start = 25.dp, end = 14.dp, bottom = 12.dp)
   305→            ) {
   306→                // Subtle separator
   307→                Box(
   308→                    modifier = Modifier
   309→                        .fillMaxWidth()
   310→                        .height(1.dp)
   311→                        .background(AppColors.SurfaceBorder)
   312→                )
   313→
   314→                Spacer(modifier = Modifier.height(8.dp))
   315→
   316→                when (event) {
   317→                    is AgentEvent.ToolExecution -> ToolExecutionDetail(event)
   318→                    is AgentEvent.Observation -> {
   319→                        Text(
   320→                            text = event.summary,
   321→                            style = MaterialTheme.typography.bodySmall,
   322→                            color = AppColors.TextSecondary,
   323→                            fontFamily = FontFamily.Monospace,
   324→                            fontSize = 12.sp,
   325→                            lineHeight = 18.sp
   326→                        )
   327→                    }
   328→                    is AgentEvent.ModelResponse -> {
   329→                        Text(
   330→                            text = event.content,
   331→                            style = MaterialTheme.typography.bodySmall,
   332→                            color = AppColors.TextSecondary,
   333→                            fontSize = 13.sp,
   334→                            lineHeight = 19.sp
   335→                        )
   336→                    }
   337→                    is AgentEvent.UserMessage -> {
   338→                        Text(
   339→                            text = event.text,
   340→                            style = MaterialTheme.typography.bodySmall,
   341→                            color = AppColors.TextSecondary,
   342→                            fontSize = 13.sp,
   343→                            lineHeight = 19.sp
   344→                        )
   345→                    }
   346→                    is AgentEvent.StatusChange -> {
   347→                        Text(
   348→                            text = "${event.from} → ${event.to}",
   349→                            style = MaterialTheme.typography.bodySmall,
   350→                            color = AppColors.TextSecondary,
   351→                            fontFamily = FontFamily.Monospace
   352→                        )
   353→                    }
   354→                    is AgentEvent.Error -> {
   355→                        Text(
   356→                            text = event.message,
   357→                            style = MaterialTheme.typography.bodySmall,
   358→                            color = AppColors.ErrorRed,
   359→                            fontSize = 13.sp,
   360→                            lineHeight = 19.sp
   361→                        )
   362→                    }
   363→                }
   364→            }
   365→        }
   366→    }
   367→}

   577→fun UserQuestionDialog(
   578→    question: String,
   579→    onAnswer: (String) -> Unit,
   580→    onDismiss: () -> Unit
   581→) {

   586→        title = {
   587→            Text(
   588→                text = "Agent Question",
   589→                color = AppColors.TextPrimary,
   590→                fontWeight = FontWeight.SemiBold
   591→            )
   592→        },
   593→        text = {

   597→                    style = MaterialTheme.typography.bodyMedium,
   598→                    color = AppColors.TextSecondary,

   606→                    singleLine = false,
