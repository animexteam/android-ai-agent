   171→@OptIn(ExperimentalMaterial3Api::class)
   172→@Composable
   173→fun EventCard(
   174→    event: AgentEvent,
   175→    expanded: Boolean,
   176→    onToggle: () -> Unit
   177→) {
   178→    val (icon, iconColor, title) = when (event) {
   179→        is AgentEvent.ToolExecution -> {
   180→            val isSuccess = event.result.contains("success", ignoreCase = true) ||
   181→                    !event.result.contains("error", ignoreCase = true)
   182→            Triple(
   183→                if (isSuccess) Icons.Default.CheckCircle else Icons.Default.Error,
   184→                if (isSuccess) AppColors.Success else AppColors.Error,
   185→                "Tool: ${event.toolName}"
   186→            )
   187→        }
   188→        is AgentEvent.Observation -> Triple(
   189→            Icons.Default.KeyboardArrowDown,
   190→            AppColors.Primary,
   191→            "Observation"
   192→        )
   193→        is AgentEvent.ModelResponse -> Triple(
   194→            Icons.AutoMirrored.Filled.ArrowRight,
   195→            AppColors.Primary,
   196→            "Model: ${event.decisionType}"
   197→        )
   198→        is AgentEvent.UserMessage -> Triple(
   199→            Icons.Default.KeyboardArrowDown,
   200→            AppColors.Primary,
   201→            "User: ${event.text.take(50)}"
   202→        )
   203→        is AgentEvent.StatusChange -> Triple(
   204→            Icons.Default.KeyboardArrowDown,
   205→            AppColors.Secondary,
   206→            "${event.from} → ${event.to}"
   207→        )
   208→        is AgentEvent.Error -> Triple(
   209→            Icons.Default.Error,
   210→            AppColors.Error,
   211→            event.message.take(80)
   212→        )
   213→    }
   214→
   215→    Card(
   216→        modifier = Modifier
   217→            .fillMaxWidth()
   218→            .padding(horizontal = 16.dp, vertical = 2.dp)
   219→            .clickable { onToggle() },
   220→        colors = CardDefaults.cardColors(
   221→            containerColor = AppColors.Surface
   222→        ),
   223→        shape = RoundedCornerShape(12.dp)
   224→    ) {
   225→        Column(
   226→            modifier = Modifier.padding(12.dp)
   227→        ) {
   228→            Row(
   229→                modifier = Modifier.fillMaxWidth(),
   230→                verticalAlignment = Alignment.CenterVertically
   231→            ) {
   232→                Icon(
   233→                    imageVector = icon,
   234→                    contentDescription = null,
   235→                    tint = iconColor,
   236→                    modifier = Modifier.size(18.dp)
   237→                )
   238→                Spacer(modifier = Modifier.width(8.dp))
   239→                Text(
   240→                    text = title,
   241→                    style = MaterialTheme.typography.bodyMedium,
   242→                    color = AppColors.TextPrimary,
   243→                    maxLines = 1,
   244→                    overflow = TextOverflow.Ellipsis,
   245→                    modifier = Modifier.weight(1f)
   246→                )
   247→                Text(
   248→                    text = "Step ${event.stepNumber}",
   249→                    style = MaterialTheme.typography.labelSmall,
   250→                    color = AppColors.TextMuted
   251→                )
   252→                Spacer(modifier = Modifier.width(4.dp))
   253→                Icon(
   254→                    imageVector = if (expanded) Icons.Default.ExpandLess else Icons.Default.ExpandMore,
   255→                    contentDescription = if (expanded) "Collapse" else "Expand",
   256→                    tint = AppColors.TextSecondary,
   257→                    modifier = Modifier.size(18.dp)
   258→                )
   259→            }
   260→
   261→            AnimatedVisibility(visible = expanded) {
   262→                Column(
   263→                    modifier = Modifier.padding(top = 8.dp)
   264→                ) {
   265→                    when (event) {
   266→                        is AgentEvent.ToolExecution -> ToolExecutionDetail(event)
   267→                        is AgentEvent.Observation -> {
   268→                            Text(
   269→                                text = event.summary,
   270→                                style = MaterialTheme.typography.bodySmall,
   271→                                color = AppColors.TextSecondary,
   272→                                fontFamily = FontFamily.Monospace
   273→                            )
   274→                        }
   275→                        is AgentEvent.ModelResponse -> {
   276→                            Text(
   277→                                text = event.content,
   278→                                style = MaterialTheme.typography.bodySmall,
   279→                                color = AppColors.TextSecondary
   280→                            )
   281→                        }
   282→                        is AgentEvent.UserMessage -> {
   283→                            Text(
   284→                                text = event.text,
   285→                                style = MaterialTheme.typography.bodySmall,
   286→                                color = AppColors.TextSecondary
   287→                            )
   288→                        }
   289→                        is AgentEvent.StatusChange -> {
   290→                            Text(
   291→                                text = "${event.from} → ${event.to}",
   292→                                style = MaterialTheme.typography.bodySmall,
   293→                                color = AppColors.TextSecondary
   294→                            )
   295→                        }
   296→                        is AgentEvent.Error -> {
   297→                            Text(
   298→                                text = event.message,
   299→                                style = MaterialTheme.typography.bodySmall,
   300→                                color = AppColors.Error
   301→                            )
   302→                        }
   303→                    }
   304→                }
   305→            }
   306→        }
   307→    }
   308→}

   406→            }
   407→        }

   590→        },

   592→        shape = RoundedCornerShape(16.dp)
   593→    )

   606→            .padding(horizontal = 16.dp, vertical = 8.dp),
   607→        colors = CardDefaults.cardColors(
   608→            containerColor = AppColors.Error.copy(alpha = 0.1f)
   609→        ),

   611→    ) {
   612→        Row(

   614→                .fillMaxWidth()
   615→                .padding(16.dp),
   616→            verticalAlignment = Alignment.CenterVertically,
   617→            horizontalArrangement = Arrangement.spacedBy(12.dp)
   618→        ) {
   619→            Icon(
   620→                imageVector = Icons.Default.SettingsAccessibility,
   621→                contentDescription = "Accessibility",
   622→                tint = AppColors.Warning,
   623→                modifier = Modifier.size(24.dp)
   624→            )
   625→            Column(
   626→                modifier = Modifier.weight(1f)
   627→            ) {
   628→                Text(
   629→                    text = "Accessibility Service Required",
   630→                    style = MaterialTheme.typography.titleSmall,
   631→                    color = AppColors.TextPrimary,

   637→                    color = AppColors.TextSecondary

   640→            Button(
   641→                onClick = onEnable,
   642→                colors = ButtonDefaults.buttonColors(
