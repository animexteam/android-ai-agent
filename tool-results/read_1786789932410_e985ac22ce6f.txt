    45→    45→@OptIn(ExperimentalMaterial3Api::class)
    46→    46→@Composable
    47→    47→fun MainScreen(
    48→    48→    viewModel: AgentViewModel,
    49→    49→    onNavigateToSettings: () -> Unit,
    50→    50→    onNavigateToDebug: () -> Unit,
    51→    51→    onNavigateToHistory: () -> Unit
    52→    52→) {
    53→    53→    val state by viewModel.stateFlow.collectAsState()
    54→    54→    val isAccessibilityEnabled by remember {
    55→    55→        mutableStateOf(viewModel.isAccessibilityServiceEnabled())
    56→    56→    }
    57→    57→    var taskInput by remember { mutableStateOf("") }
    58→    58→    val listState = rememberLazyListState()
    59→    59→    val isRunning = state.status.isActive
    60→    60→    val context = LocalContext.current
    61→    61→
    62→    62→    var isListening by remember { mutableStateOf(false) }
    63→    63→
    64→    64→    val displayItems = remember(state.history) {
    65→    65→        state.history.mapNotNull { event ->
    66→    66→            when (event) {
    67→    67→                is AgentEvent.UserMessage -> DisplayItem.UserMessage(event.text, event.timestamp)
    68→    68→                is AgentEvent.AgentMessage -> DisplayItem.AgentMessage(event.text, event.timestamp)
    69→    69→                is AgentEvent.ToolExecution -> DisplayItem.ToolAction(
    70→    70→                    name = event.toolName,
    71→    71→                    args = event.arguments,
    72→    72→                    success = event.result.success,
    73→    73→                    error = event.result.error?.message
    74→    74→                )
    75→    75→                is AgentEvent.Error -> DisplayItem.ErrorItem(event.message)
    76→    76→                else -> null
    77→    77→            }
    78→    78→        }
    79→    79→    }
    80→    80→
    81→    81→    LaunchedEffect(displayItems.size) {
    82→    82→        if (displayItems.isNotEmpty()) {
    83→    83→            listState.animateScrollToItem(displayItems.size - 1)
    84→    84→        }
    85→    85→    }
    86→    86→
    87→    87→    val speechRecognizer = remember {
    88→    88→        try { SpeechRecognizer.createSpeechRecognizer(context) } catch (_: Exception) { null }
    89→    89→    }
    90→    90→    DisposableEffect(Unit) { onDispose { speechRecognizer?.destroy() } }
    91→    91→
    92→    92→    fun startVoiceInput() {
    93→    93→        if (speechRecognizer == null) return
    94→    94→        isListening = true
    95→    95→        val intent = Intent(RecognizerIntent.ACTION_RECOGNIZE_SPEECH).apply {
    96→    96→            putExtra(RecognizerIntent.EXTRA_LANGUAGE_MODEL, RecognizerIntent.LANGUAGE_MODEL_FREE_FORM)
    97→    97→            putExtra(RecognizerIntent.EXTRA_LANGUAGE, "en-IN")
    98→    98→            putExtra(RecognizerIntent.EXTRA_PROMPT, "Speak to Android-Use...")
    99→    99→        }
   100→   100→        try {
   101→   101→            speechRecognizer.setRecognitionListener(object : RecognitionListener {
   102→   102→                override fun onReadyForSpeech(p: Bundle?) {}
   103→   103→                override fun onBeginningOfSpeech() {}
   104→   104→                override fun onRmsChanged(r: Float) {}
   105→   105→                override fun onBufferReceived(b: ByteArray?) {}
   106→   106→                override fun onEndOfSpeech() { isListening = false }
   107→   107→                override fun onError(e: Int) { isListening = false }
   108→   108→                override fun onResults(results: Bundle?) {
   109→   109→                    val matches = results?.getStringArrayList(SpeechRecognizer.RESULTS_RECOGNITION)
   110→   110→                    if (!matches.isNullOrEmpty()) taskInput = matches.first()
   111→   111→                    isListening = false
   112→   112→                }
   113→   113→                override fun onPartialResults(p: Bundle?) {}
   114→   114→                override fun onEvent(e: Int, p: Bundle?) {}
   115→   115→            })
   116→   116→            speechRecognizer.startListening(intent)
   117→   117→        } catch (_: ActivityNotFoundException) { isListening = false }
   118→   118→    }
   119→   119→
   120→   120→    fun sendTask(text: String) {
   121→   121→        val trimmed = text.trim()
   122→   122→        if (trimmed.isNotBlank()) {
   123→   123→            if (state.goal.isBlank()) viewModel.startTask(trimmed)
   124→   124→            else viewModel.continueChat(trimmed)
   125→   125→            taskInput = ""
   126→   126→        }
   127→   127→    }
   128→   128→
   129→   129→    Box(modifier = Modifier.fillMaxSize().background(AppColors.DarkBackground)) {
   130→   130→        Column(modifier = Modifier.fillMaxSize().align(Alignment.TopStart)) {
   131→   131→            TopBar(
   132→   132→                onNewChat = { viewModel.newChat() },
   133→   133→                onHistory = onNavigateToHistory,
   134→   134→                onSettings = onNavigateToSettings,
   135→   135→                onDebug = onNavigateToDebug
   136→   136→            )
   137→   137→            if (isRunning) RunningStatusBar(state = state)
   138→   138→            if (!isAccessibilityEnabled) {
   139→   139→                AccessibilityBanner(onEnable = { viewModel.openAccessibilitySettings() })
   140→   140→            }
   141→   141→            if (state.status == AgentStatus.COMPLETED && !isRunning) {
   142→   142→                ResultBanner(text = "Done", isSuccess = true)
   143→   143→            } else if (state.status == AgentStatus.FAILED && !isRunning) {
   144→   144→                ResultBanner(text = state.lastError ?: "Something went wrong", isSuccess = false)
   145→   145→            } else if (state.status == AgentStatus.CANCELLED && !isRunning) {
   146→   146→                ResultBanner(text = "Stopped", isSuccess = false)
   147→   147→            }
   148→   148→
   149→   149→            if (displayItems.isEmpty() && !isRunning) {
   150→   150→                EmptyState(onSuggestionClick = { sendTask(it) })
   151→   151→            } else {
   152→   152→                LazyColumn(
   153→   153→                    state = listState,
   154→   154→                    modifier = Modifier.fillMaxSize().padding(horizontal = 14.dp),
   155→   155→                    verticalArrangement = Arrangement.spacedBy(5.dp),
   156→   156→                    contentPadding = PaddingValues(vertical = 6.dp)
   157→   157→                ) {
   158→   158→                    items(items = displayItems, key = { it.timestamp.toString() + it::class.simpleName }) { item ->
   159→   159→                        when (item) {
   160→   160→                            is DisplayItem.UserMessage -> UserChatBubble(text = item.text)
   161→   161→                            is DisplayItem.AgentMessage -> AgentChatBubble(text = item.text)
   162→   162→                            is DisplayItem.ToolAction -> ToolActionCard(item)
   163→   163→                            is DisplayItem.ErrorItem -> ErrorText(item.message)
   164→   164→                        }
   165→   165→                    }
   166→   166→                    item { Spacer(modifier = Modifier.height(90.dp)) }
   167→   167→                }
   168→   168→            }
   169→   169→        }
   170→   170→
   171→   171→        // ── BOTTOM INPUT BAR — anchored to bottom ──
   172→   172→        Box(
   173→   173→            modifier = Modifier
   174→   174→                .align(Alignment.BottomCenter)
   175→   175→                .fillMaxWidth()
   176→   176→                .background(AppColors.DarkBackground)
   177→   177→        ) {
   178→   178→            BottomInputBar(
   179→   179→                input = taskInput,
   180→   180→                onInputChange = { taskInput = it },
   181→   181→                onSend = { sendTask(taskInput) },
   182→   182→                onStop = { viewModel.stopAgent() },
   183→   183→                isRunning = isRunning,
   184→   184→                isListening = isListening,
   185→   185→                onVoiceClick = { startVoiceInput() },
   186→   186→                canSend = taskInput.isNotBlank() && !isRunning
   187→   187→            )
   188→   188→        }
   189→   189→
   190→   190→        state.pendingConfirmation?.let { conf ->
   191→   191→            ConfirmationDialog(conf, onConfirm = { viewModel.respondToConfirmation(true) }, onDeny = { viewModel.respondToConfirmation(false) })
   192→   192→        }
   193→   193→        state.pendingQuestion?.let { q ->
   194→   194→            UserQuestionDialog(q, onAnswer = { viewModel.respondToUser(it) }, onDismiss = {})
   195→   195→        }
   196→   196→    }
   197→   197→}

   273→   273→        modifier = Modifier.fillMaxWidth().background(color.copy(alpha = 0.06f)).padding(horizontal = 14.dp, vertical = 6.dp),
   274→   274→        verticalAlignment = Alignment.CenterVertically
   275→   275→    ) {
   276→   276→        Icon(icon, null, tint = color, modifier = Modifier.size(14.dp).alpha(pulse.value))
   277→   277→        Spacer(Modifier.width(8.dp))
   278→   278→        Text(label, color = color, fontSize = 12.5.sp, fontWeight = FontWeight.Medium, modifier = Modifier.weight(1f))
   279→   279→        Text("${state.durationFormatted}", color = AppColors.TextMuted, fontSize = 11.sp)
   280→   280→        Spacer(Modifier.width(10.dp))
   281→   281→        Text("Step ${state.stepNumber}", color = AppColors.TextMuted, fontSize = 11.sp)
   282→   282→    }
   283→   283→}
