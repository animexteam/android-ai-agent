     2→
     3→import android.util.Log
     4→import com.androidagent.aiagent.ai.GemmaClient

    11→import kotlinx.coroutines.CoroutineScope
    12→import kotlinx.coroutines.Dispatchers
    13→import kotlinx.coroutines.Job
    14→import kotlinx.coroutines.SupervisorJob

    20→
    21→/**
    22→ * The heart of the agent – runs the observe → reason → act → verify loop.
    23→ *
    24→ * Architecture:
    25→ * - The runtime owns a single [MutableStateFlow] of [AgentState] that the
    26→ *   UI observes.  All state mutations go through `copy()` / `withEvent()`.
    27→ * - The main loop ([runAgentLoop]) is launched in a dedicated
    28→ *   [SupervisorJob] + [Dispatchers.Main] scope so that the UI stays
    29→ *   responsive.
    30→ * - Prompt construction is **entirely delegated** to [AgentPromptBuilder].
    31→ * - Tool name resolution (aliases, fuzzy match) is handled by
    32→ *   [ToolExecutor.resolveToolName].
    33→ * - Loop detection is delegated to [AgentLoopGuard].
    34→ */
    35→class AgentRuntime(
    36→    private val gemmaClient: GemmaClient,
    37→    private val toolRegistry: ToolRegistry,
    38→    private val toolExecutor: ToolExecutor,
    39→    private val accessibilityObserver: AccessibilityObserver,
    40→    private val settingsRepository: SettingsRepository,
    41→    private val taskRepository: TaskRepository? = null,
    42→    private val promptBuilder: AgentPromptBuilder = AgentPromptBuilder(),
    43→    private val loopGuard: AgentLoopGuard = AgentLoopGuard()
    44→) {
    45→
    46→    companion object {
    47→        private const val TAG = "AgentRuntime"
    48→        /** Maximum model call retries on transient errors. */
    49→        private const val MAX_MODEL_RETRIES = 2
    50→    }
    51→
    52→    // -------------------------------------------------------------------
    53→    // State
    54→    // -------------------------------------------------------------------
    55→
    56→    private val _state = MutableStateFlow(AgentState())
    57→    val state: StateFlow<AgentState> = _state.asStateFlow()
    58→
    59→    private var agentJob: Job? = null
    60→    private val agentScope = CoroutineScope(SupervisorJob() + Dispatchers.Main)
    61→
    62→    /**
    63→     * System prompt is built once per task (not every turn) because it
    64→     * contains the tool catalogue which doesn't change during a task.
    65→     */
    66→    private var cachedSystemPrompt: String? = null
    67→
    68→    /**
    69→     * Known tool names, refreshed when the system prompt is built.
    70→     * Passed to [DecisionParser] so it can validate tool names.
    71→     */
    72→    private var knownToolNames: Set<String> = emptySet()
    73→
    74→    // -------------------------------------------------------------------
    75→    // Public API
    76→    // -------------------------------------------------------------------
    77→
    78→    /** Start a new task, cancelling any running task first. */
    79→    fun startTask(goal: String) {
    80→        agentJob?.cancel()
    81→        loopGuard.reset()
    82→        cachedSystemPrompt = null
    83→
    84→        _state.value = AgentState(
    85→            goal = goal,
    86→            status = AgentStatus.THINKING,
    87→            stepNumber = 0,
    88→            startTime = System.currentTimeMillis()
    89→        )
    90→
    91→        agentJob = agentScope.launch {
    92→            runAgentLoop()
    93→        }
    94→    }
    95→
    96→    /** Resume the agent after the user answered a question. */
    97→    fun respondToUser(answer: String) {
    98→        val currentState = _state.value
    99→        if (currentState.status != AgentStatus.WAITING_FOR_USER) return
   100→
   101→        val userEvent = AgentEvent.UserMessage(
   102→            stepNumber = currentState.stepNumber,
   103→            text = answer
   104→        )
   105→
   106→        _state.value = currentState
   107→            .withEvent(userEvent)
   108→            .copy(
   109→                pendingQuestion = null,
   110→                status = AgentStatus.THINKING
   111→            )
   112→
   113→        agentJob = agentScope.launch { runAgentLoop() }
   114→    }
   115→
   116→    /** Resume the agent after the user responded to a safety confirmation. */
   117→    fun respondToConfirmation(confirmed: Boolean) {
   118→        val currentState = _state.value
   119→        if (currentState.status != AgentStatus.WAITING_FOR_CONFIRMATION) return
   120→        val pending = currentState.pendingConfirmation ?: return
   121→
   122→        if (confirmed) {
   123→            agentJob = agentScope.launch {
   124→                try {
   125→                    val observationId = currentState.currentObservationId
   126→                    val argsJson = try {
   127→                        kotlinx.serialization.json.Json.parseToJsonElement(pending.arguments)
   128→                    } catch (_: Exception) { null }
   129→                    val args = argsJson as? kotlinx.serialization.json.JsonObject
   130→                        ?: kotlinx.serialization.json.JsonObject(emptyMap())
   131→
   132→                    val toolResult = withContext(Dispatchers.IO) {
   133→                        toolExecutor.execute(pending.toolName, args, observationId)
   134→                    }
   135→
   136→                    val toolEvent = AgentEvent.ToolExecution(
   137→                        stepNumber = currentState.stepNumber,
   138→                        toolName = pending.toolName,
   139→                        arguments = pending.arguments,
   140→                        result = toolResult
   141→                    )
   142→
   143→                    loopGuard.recordAction(pending.toolName, pending.arguments)
   144→
   145→                    _state.value = _state.value
   146→                        .withEvent(toolEvent)
   147→                        .copy(
   148→                            pendingConfirmation = null,
   149→                            status = AgentStatus.THINKING,
   150→                            stepNumber = _state.value.stepNumber + 1
   151→                        )
   152→
   153→                    runAgentLoop()
   154→                } catch (e: CancellationException) {
   155→                    throw e
   156→                } catch (e: Exception) {
   157→                    Log.e(TAG, "Failed to execute confirmed action", e)
   158→                    _state.value = _state.value.copy(
   159→                        pendingConfirmation = null,
   160→                        status = AgentStatus.THINKING,
   161→                        lastError = e.message
   162→                    )
   163→                    runAgentLoop()
   164→                }
   165→            }
   166→        } else {
   167→            val errorEvent = AgentEvent.Error(
   168→                stepNumber = currentState.stepNumber,
   169→                message = "User rejected the action: ${pending.toolName}"
   170→            )
   171→            _state.value = currentState
   172→                .withEvent(errorEvent)
   173→                .copy(
   174→                    pendingConfirmation = null,
   175→                    status = AgentStatus.THINKING
   176→                )
   177→
   178→            agentJob = agentScope.launch { runAgentLoop() }
   179→        }
   180→    }
   181→
   182→    /** Cancel the current task. */
   183→    fun stopAgent() {
   184→        Log.i(TAG, "Stop requested, cancelling agent job")
   185→        agentJob?.cancel()
   186→        agentJob = null
   187→        _state.value = _state.value.copy(
   188→            status = AgentStatus.CANCELLED,
   189→            endTime = System.currentTimeMillis(),
   190→            pendingConfirmation = null,
   191→            pendingQuestion = null
   192→        )
   193→    }
   194→
   195→    // ===================================================================
   196→    // Main agent loop
   197→    // ===================================================================
   198→
   199→    private suspend fun runAgentLoop() {
   200→        try {
   201→            val maxSteps = withContext(Dispatchers.IO) {
   202→                settingsRepository.maxSteps()
   203→            }
   204→            _state.value = _state.value.copy(maxSteps = maxSteps)
   205→
   206→            // Build system prompt once (includes current tool catalogue)
   207→            ensureSystemPrompt()
   208→
   209→            while (_state.value.status.isActive) {
   210→                val current = _state.value
   211→
   212→                // --- Step budget check ---
   213→                if (current.stepNumber >= maxSteps) {
   214→                    finishTask(
   215→                        status = AgentStatus.FAILED,
   216→                        error = "Max steps ($maxSteps) reached without completing the task."
   217→                    )
   218→                    break
   219→                }
   220→
   221→                // --- Observe ---
   222→                val observation = observeScreen()
   223→                if (observation == null) continue  // error already recorded
   224→
   225→                // --- Loop detection ---
   226→                val loopWarning = if (loopGuard.isLooping()) {
   227→                    loopGuard.getLoopMessage()
   228→                } else null
   229→
   230→                // --- Call model ---
   231→                val decision = callModel(observation, loopWarning)
   232→                    ?: continue  // model error already recorded, step consumed
   233→
   234→                // --- Record model event ---
   235→                val modelEvent = AgentEvent.ModelResponse(
   236→                    stepNumber = _state.value.stepNumber,
   237→                    decisionType = decision.type,
   238→                    content = describeDecision(decision)
   239→                )
   240→                _state.value = _state.value.withEvent(modelEvent)
   241→
   242→                // --- Execute decision ---
   243→                val shouldContinue = executeDecision(decision, observation.id)
   244→                if (!shouldContinue) break
   245→
   246→                // --- Advance step ---
   247→                _state.value = _state.value.copy(
   248→                    stepNumber = _state.value.stepNumber + 1
   249→                )
   250→            }
   251→        } catch (e: CancellationException) {
   252→            Log.i(TAG, "Agent loop cancelled")
   253→        } catch (e: Exception) {
   254→            Log.e(TAG, "Agent loop error", e)
   255→            finishTask(
   256→                status = AgentStatus.FAILED,
   257→                error = e.message ?: "Unknown error"
   258→            )
   259→        }
   260→    }
   261→
   262→    // ===================================================================
   263→    // Step 1: Observe
   264→    // ===================================================================
   265→
   266→    /**
   267→     * Observe the screen and update state.  Returns the observation,
   268→     * or null if observation failed (in which case an error event is
   269→     * recorded and the step is consumed).
   270→     */
   271→    private suspend fun observeScreen(): AndroidObservation? {
   272→        _state.value = _state.value.copy(status = AgentStatus.THINKING)
   273→
   274→        val useVision = shouldUseVision()
   275→        val observation: AndroidObservation
   276→        try {
   277→            observation = withContext(Dispatchers.IO) {
   278→                accessibilityObserver.observeWithScreenshot(takeScreenshot = useVision)
   279→            }
   280→        } catch (e: CancellationException) {
   281→            throw e
   282→        } catch (e: Exception) {
   283→            Log.w(TAG, "Observation failed: ${e.message}")
   284→            _state.value = _state.value
   285→                .withEvent(
   286→                    AgentEvent.Error(
   287→                        stepNumber = _state.value.stepNumber,
   288→                        message = "Failed to observe screen: ${e.message}"
   289→                    )
   290→                )
   291→                .copy(stepNumber = _state.value.stepNumber + 1)
   292→            return null
   293→        }
   294→
   295→        val obsEvent = AgentEvent.Observation(
   296→            stepNumber = _state.value.stepNumber,
   297→            summary = "Package: ${observation.packageName}, " +
   298→                "Activity: ${observation.activityName}, " +
   299→                "Nodes: ${observation.uiTree.size}, " +
   300→                "Has screenshot: ${observation.screenshotBase64 != null}"
   301→        )
   302→
   303→        loopGuard.recordObservation(observation)
   304→
   305→        _state.value = _state.value
   306→            .withEvent(obsEvent)
   307→            .copy(
   308→                lastObservation = observation,
   309→                currentPackage = observation.packageName,
   310→                currentObservationId = observation.id
   311→            )
   312→
   313→        return observation
   314→    }
   315→
   316→    // ===================================================================
   317→    // Step 2: Call model
   318→    // ===================================================================
   319→
   320→    /**
   321→     * Call the AI model with the current screen state and history.
   322→     * Returns the parsed decision, or null if the call failed (error
   323→     * is recorded and step consumed).
   324→     */
   325→    private suspend fun callModel(
   326→        observation: AndroidObservation,
   327→        loopWarning: String?
   328→       ): AgentDecision? {
   329→        val current = _state.value
   330→        val systemPrompt = cachedSystemPrompt!!
   331→
   332→        val userMessage = promptBuilder.buildUserMessage(
   333→            goal = current.goal,
   334→            observation = observation,
   335→            history = current.history,
   336→            loopWarning = loopWarning
   337→        )
   338→
   339→        val screenshotBase64 = observation.screenshotBase64
   340→        var lastError: String? = null
   341→
   342→        // Retry transient model failures up to MAX_MODEL_RETRIES times.
   343→        for (attempt in 0..MAX_MODEL_RETRIES) {
   344→            val callStartTime = System.currentTimeMillis()
   345→            try {
   346→                val response = withContext(Dispatchers.IO) {
   347→                    gemmaClient.generate(
   348→                        systemPrompt = systemPrompt,
   349→                        userMessage = userMessage,
   350→                        tools = toolRegistry.getAll(),
   351→                        screenshotBase64 = screenshotBase64
   352→                    )
   353→                }
   354→
   355→                _state.value = _state.value.copy(
   356→                    modelLatencyMs = System.currentTimeMillis() - callStartTime
   357→                )
   358→
   359→                return DecisionParser.parse(response, knownToolNames)
   360→
   361→            } catch (e: CancellationException) {
   362→                throw e
   363→            } catch (e: Exception) {
   364→                lastError = e.message
   365→                Log.w(TAG, "Model call failed (attempt ${attempt + 1}/$MAX_MODEL_RETRIES): ${e.message}")
   366→                if (attempt == MAX_MODEL_RETRIES) break
   367→                // Brief pause before retry
   368→                kotlinx.coroutines.delay(500L * (attempt + 1))
   369→            }
   370→        }
   371→
   372→        // All retries exhausted
   373→        Log.w(TAG, "Model call failed after all retries: $lastError")
   374→        _state.value = _state.value
   375→            .withEvent(
   376→                AgentEvent.Error(
   377→                    stepNumber = _state.value.stepNumber,
   378→                    message = "Model call failed: $lastError"
   379→                )
   380→            )
   381→            .copy(
   382→                lastError = "Model error: $lastError",
   383→                stepNumber = _state.value.stepNumber + 1
   384→            )
   385→        return null
   386→    }
   387→
   388→    // ===================================================================
   389→    // Step 3: Execute decision
   390→    // ===================================================================
   391→
   392→    /**
   393→     * Execute the parsed model decision.  Returns `true` if the loop
   394→     * should continue, `false` if the task is done or paused.
   395→     */
   396→    private suspend fun executeDecision(
   397→        decision: AgentDecision,
   398→        observationId: String
   399→    ): Boolean {
   400→        return when (decision) {
   401→            is ToolCallDecision -> executeToolCall(decision, observationId)
   402→            is AskUserDecision -> {
   403→                _state.value = _state.value.copy(
   404→                    status = AgentStatus.WAITING_FOR_USER,
   405→                    pendingQuestion = decision.question
   406→                )
   407→                false  // pause loop
   408→            }
   409→            is FinishDecisionData -> {
   410→                val newStatus = if (decision.success) AgentStatus.COMPLETED else AgentStatus.FAILED
   411→                finishTask(
   412→                    status = newStatus,
   413→                    error = if (!decision.success) decision.message else null
   414→                )
   415→                false  // stop loop
   416→            }
   417→            is MessageDecision -> {
   418→                // Status message – nothing to do, loop continues.
   419→                true
   420→            }
   421→            is ErrorDecisionData -> {
   422→                _state.value = _state.value
   423→                    .withEvent(
   424→                        AgentEvent.Error(
   425→                            stepNumber = _state.value.stepNumber,
   426→                            message = decision.message
   427→                        )
   428→                    )
   429→                    .copy(lastError = decision.message)
   430→                true  // loop continues so the model can recover
   431→            }
   432→        }
   433→    }
   434→
   435→    // -------------------------------------------------------------------
   436→    // Tool call execution
   437→    // -------------------------------------------------------------------
   438→
   439→    private suspend fun executeToolCall(
   440→        decision: ToolCallDecision,
   441→        observationId: String
   442→    ): Boolean {
   443→        _state.value = _state.value.copy(status = AgentStatus.EXECUTING)
   444→
   445→        // Resolve aliases before execution
   446→        val resolution = toolExecutor.resolveToolName(decision.toolName)
   447→        val toolName = resolution.resolvedName
   448→
   449→        // If the model used a wrong name, log it but execute the corrected name.
   450→        if (resolution.wasCorrected) {
   451→            Log.i(TAG, "Auto-corrected tool name: ${decision.toolName} → $toolName (${resolution.method})")
   452→        }
   453→
   454→        // Execute (safety checks are inside ToolExecutor)
   455→        var toolResult: ToolResult
   456→        try {
   457→            toolResult = withContext(Dispatchers.IO) {
   458→                toolExecutor.execute(toolName, decision.arguments, observationId)
   459→            }
   460→        } catch (e: CancellationException) {
   461→            throw e
   462→        } catch (e: Exception) {
   463→            Log.e(TAG, "Tool execution exception for $toolName", e)
   464→            _state.value = _state.value
   465→                .withEvent(
   466→                    AgentEvent.Error(
   467→                        stepNumber = _state.value.stepNumber,
   468→                        message = "Tool execution exception for $toolName: ${e.message}"
   469→                    )
   470→                    )
   471→                .copy(status = AgentStatus.THINKING, lastError = e.message)
   472→            return true
   473→        }
   474→
   475→        // Handle UNKNOWN_TOOL (should be rare since we resolve aliases first)
   476→        if (!toolResult.success && toolResult.error?.code == ToolExecutor.ERROR_UNKNOWN_TOOL) {
   477→            _state.value = _state.value
   478→                .withEvent(
   479→                    AgentEvent.Error(
   480→                        stepNumber = _state.value.stepNumber,
   481→                        message = buildString {
   482→                            append("Tool '$toolName' does not exist. ")
   483→                            append("Available: ${toolExecutor.getRegisteredHandlerNames().sorted().joinToString(", ")}.")
   484→                            append(" Please use a valid tool name.")
   485→                        }
   486→                    )
   487→                )
   488→                .copy(status = AgentStatus.THINKING, lastError = "Unknown tool: $toolName")
   489→            return true
   490→        }
   491→
   492→        // Handle NODE_NOT_FOUND with automatic re-observe + retry
   493→        if (!toolResult.success && toolResult.error?.code == "NODE_NOT_FOUND") {
   494→            toolResult = retryAfterReobserve(decision, toolResult)
   495→        }
   496→
   497→        // Record the tool execution event
   498→        val toolEvent = AgentEvent.ToolExecution(
   499→            stepNumber = _state.value.stepNumber,
   500→            toolName = toolName,
   501→            arguments = decision.arguments.toString(),
   502→            result = toolResult
   503→        )
   504→        loopGuard.recordAction(toolName, decision.arguments.toString())
   505→        _state.value = _state.value.withEvent(toolEvent)
   506→
   507→        // Handle post-execution status transitions
   508→        return handlePostExecutionStatus(toolResult, decision)
   509→    }
   510→
   511→    /**
   512→     * When a NODE_NOT_FOUND error occurs, re-observe the screen (without
   513→     * screenshot) and retry the same tool call once.
   514→     */
   515→    private suspend fun retryAfterReobserve(
   516→        decision: ToolCallDecision,
   517→        originalResult: ToolResult
   518→    ): ToolResult {
   519→        Log.i(TAG, "NODE_NOT_FOUND for ${decision.toolName}, re-observing and retrying once")
   520→
   521→        return try {
   522→            val newObservation = withContext(Dispatchers.IO) {
   523→                accessibilityObserver.observeWithScreenshot(takeScreenshot = false)
   524→            }
   525→
   526→            val retryObsEvent = AgentEvent.Observation(
   527→                stepNumber = _state.value.stepNumber,
   528→                summary = "Auto re-observe (NODE_NOT_FOUND retry): " +
   529→                    "Package: ${newObservation.packageName}, Nodes: ${newObservation.uiTree.size}"
   530→            )
   531→            loopGuard.recordObservation(newObservation)
   532→
   533→            _state.value = _state.value
   534→                .withEvent(retryObsEvent)
   535→                .copy(
   536→                    lastObservation = newObservation,
   537→                    currentObservationId = newObservation.id
   538→                )
   539→
   540→            withContext(Dispatchers.IO) {
   541→                toolExecutor.execute(decision.toolName, decision.arguments, newObservation.id)
   542→            }
   543→        } catch (e: CancellationException) {
   544→            throw e
   545→        } catch (e: Exception) {
   546→            Log.w(TAG, "Re-observation/retry failed, using original NODE_NOT_FOUND result", e)
   547→            originalResult
   548→        }
   549→    }
   550→
   551→    /**
   552→     * After a tool call completes, transition the agent's status based
   553→     * on the result's error code.
   554→     */
   555→    private fun handlePostExecutionStatus(
   556→        toolResult: ToolResult,
   557→        decision: ToolCallDecision
   558→    ): Boolean {
   559→        return when {
   560→            toolResult.error?.code == "NODE_STALE" -> {
   561→                _state.value = _state.value.copy(
   562→                    status = AgentStatus.THINKING,
   563→                    lastError = "Node was stale, will re-observe"
   564→                )
   565→                true
   566→            }
   567→            toolResult.error?.code == ToolExecutor.ERROR_CONFIRMATION_REQUIRED -> {
   568→                _state.value = _state.value.copy(
   569→                    status = AgentStatus.WAITING_FOR_CONFIRMATION,
   570→                    pendingConfirmation = PendingConfirmation(
   571→                        toolName = decision.toolName,
   572→                        arguments = decision.arguments.toString(),
   573→                        reason = toolResult.error.message,
   574→                        stepNumber = _state.value.stepNumber
   575→                    )
   576→                )
   577→                false  // pause loop
   578→            }
   579→            toolResult.error?.code == ToolExecutor.ERROR_BLOCKED -> {
   580→                _state.value = _state.value
   581→                    .withEvent(
   582→                        AgentEvent.Error(
   583→                            stepNumber = _state.value.stepNumber,
   584→                            message = "Action blocked: ${toolResult.error.message}"
   585→                        )
   586→                    )
   587→                    .copy(
   588→                        status = AgentStatus.THINKING,
   589→                        lastError = toolResult.error.message
   590→                    )
   591→                true
   592→            }
   593→            !toolResult.success -> {
   594→                _state.value = _state.value.copy(
   595→                    status = AgentStatus.THINKING,
   596→                    lastError = toolResult.error?.message ?: "Tool failed"
   597→                )
   598→                true
   599→            }
   600→            else -> {
   601→                _state.value = _state.value.copy(status = AgentStatus.THINKING)
   602→                true
   603→            }
   604→        }
   605→    }

   611→    /**
   612→     * Ensure the cached system prompt is built with the current tool set.
   613→     * Called once at the start of each task.
   614→     */
   615→    private fun ensureSystemPrompt() {
   616→        if (cachedSystemPrompt == null) {
   617→            val tools = toolRegistry.getAll()
   618→            knownToolNames = toolRegistry.getToolNames()
   619→            cachedSystemPrompt = promptBuilder.buildSystemPrompt(tools)
   620→            Log.d(TAG, "System prompt built with ${tools.size} tools")
   621→        }
   622→    }
   623→
   624→    /** Determine whether to include a screenshot based on vision mode settings. */
   625→    private suspend fun shouldUseVision(): Boolean {
   626→        return withContext(Dispatchers.IO) {
   627→            when (settingsRepository.visionMode()) {
   628→                "ALWAYS" -> true
   629→                "OFF" -> false
   630→                "AUTO" -> {
   631→                    val lastObs = _state.value.lastObservation
   632→                    val actionableCount = lastObs?.uiTree
   633→                        ?.count { it.isClickable || it.isEditable } ?: 0
   634→                    val lastToolFailed = _state.value.history
   635→                        .filterIsInstance<AgentEvent.ToolExecution>()
   636→                        .lastOrNull()
   637→                        ?.let { !it.result.success } ?: false
   638→                    actionableCount < 3 || lastToolFailed
   639→                }
   640→                else -> false
   641→            }
   642→        }
   643→    }
   644→
   645→    /** Produce a one-line summary of a decision for the history log. */
   646→    private fun describeDecision(decision: AgentDecision): String {
   647→        return when (decision) {
   648→            is ToolCallDecision  -> "tool_call: ${decision.toolName}"
   649→            is MessageDecision   -> decision.content
   650→            is AskUserDecision   -> "ask_user: ${decision.question}"
   651→            is FinishDecisionData -> "finish: ${decision.message}"
   652→            is ErrorDecisionData  -> "error: ${decision.message}"
   653→        }
   654→    }
   655→
   656→    /** Transition the agent to a terminal state and persist the task. */
   657→    private fun finishTask(status: AgentStatus, error: String?) {
   658→        _state.value = _state.value.copy(
   659→            status = status,
   660→            endTime = System.currentTimeMillis(),
   661→            lastError = error
   662→        )
   663→        saveTask()
   664→    }
   665→
   666→    /** Persist the completed/failed task to the database (fire-and-forget). */
   667→    private fun saveTask() {
   668→        val currentState = _state.value
   669→        taskRepository?.let { repo ->
   670→            agentScope.launch {
   671→                try {
   672→                    withContext(Dispatchers.IO) {
   673→                        val record = com.androidagent.aiagent.data.TaskRecord(
   674→                            goal = currentState.goal,
   675→                            status = currentState.status.name,
   676→                            startTime = currentState.startTime ?: System.currentTimeMillis(),
   677→                            endTime = currentState.endTime,
   678→                            stepCount = currentState.stepNumber,
   679→                            result = if (currentState.status == AgentStatus.COMPLETED)
   680→                                "Completed" else currentState.lastError
   681→                        )
   682→                        repo.insertTask(record)
   683→                    }
   684→                } catch (_: Exception) { /* best-effort persistence */ }
   685→            }
   686→        }
   687→    }
   688→}
