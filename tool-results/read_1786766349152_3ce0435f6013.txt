    43→@OptIn(ExperimentalMaterial3Api::class)
    44→@Composable
    45→fun MainScreen(
    46→    viewModel: AgentViewModel,
    47→    onNavigateToSettings: () -> Unit,
    48→    onNavigateToDebug: () -> Unit,
    49→    onNavigateToHistory: () -> Unit
    50→) {
    51→    val state by viewModel.stateFlow.collectAsState()
    52→    val isAccessibilityEnabled by remember {
    53→        mutableStateOf(viewModel.isAccessibilityServiceEnabled())
    54→    }
    55→    var taskInput by remember { mutableStateOf("") }
    56→    val listState = rememberLazyListState()
    57→    val isRunning = state.status.isActive
    58→    val context = LocalContext.current
    59→
    60→    // Voice input state
    61→    var isListening by remember { mutableStateOf(false) }
    62→    var voiceText by remember { mutableStateOf("") }
    63→
    64→    // Build display items: chat messages + tool actions
    65→    val displayItems = remember(state.history) {
    66→        state.history.mapNotNull { event ->
    67→            when (event) {
    68→                is AgentEvent.UserMessage -> DisplayItem.UserMessage(event.text, event.timestamp)
    69→                is AgentEvent.AgentMessage -> DisplayItem.AgentMessage(event.text, event.timestamp)
    70→                is AgentEvent.ToolExecution -> DisplayItem.ToolAction(
    71→                    name = event.toolName,
    72→                    args = event.arguments,
    73→                    success = event.result.success,
    74→                    error = event.result.error?.message
    75→                )
    76→                is AgentEvent.Error -> DisplayItem.ErrorItem(event.message)
    77→                else -> null
    78→            }
    79→        }
    80→    }
    81→
    82→    LaunchedEffect(displayItems.size) {
    83→        if (displayItems.isNotEmpty()) {
    84→            listState.animateScrollToItem(displayItems.size - 1)
    85→        }
    86→    }
    87→
    88→    // Voice recognizer
    89→    val speechRecognizer = remember {
    90→        try {
    91→            SpeechRecognizer.createSpeechRecognizer(context)
    92→        } catch (e: Exception) { null }
    93→    }
    94→
    95→    DisposableEffect(Unit) {
    96→        onDispose {
    97→            speechRecognizer?.destroy()
    98→        }
    99→    }
   100→
   101→    fun startVoiceInput() {
   102→        if (speechRecognizer == null) return
   103→        isListening = true
   104→        voiceText = ""
   105→        val intent = Intent(RecognizerIntent.ACTION_RECOGNIZE_SPEECH).apply {
   106→            putExtra(RecognizerIntent.EXTRA_LANGUAGE_MODEL, RecognizerIntent.LANGUAGE_MODEL_FREE_FORM)
   107→            putExtra(RecognizerIntent.EXTRA_LANGUAGE, "en-IN")
   108→            putExtra(RecognizerIntent.EXTRA_PROMPT, "Speak to Android-Use...")
   109→        }
   110→        try {
   111→            speechRecognizer.setRecognitionListener(object : RecognitionListener {
   112→                override fun onReadyForSpeech(params: Bundle?) {}
   113→                override fun onBeginningOfSpeech() {}
   114→                override fun onRmsChanged(rmsdB: Float) {}
   115→                override fun onBufferReceived(buffer: ByteArray?) {}
   116→                override fun onEndOfSpeech() {
   117→                    isListening = false
   118→                }
   119→                override fun onError(error: Int) {
   120→                    isListening = false
   121→                }
   122→                override fun onResults(results: Bundle?) {
   123→                    val matches = results?.getStringArrayList(SpeechRecognizer.RESULTS_RECOGNITION)
   124→                    val text = matches?.firstOrNull() ?: ""
   125→                    if (text.isNotBlank()) {
   126→                        taskInput = text
   127→                    }
   128→                    isListening = false
   129→                }
   130→                override fun onPartialResults(partialResults: Bundle?) {
   131→                    val matches = partialResults?.getStringArrayList(SpeechRecognizer.RESULTS_RECOGNITION)
   132→                    voiceText = matches?.firstOrNull() ?: ""
   133→                }
   134→                override fun onEvent(eventType: Int, params: Bundle?) {}
   135→            })
   136→            speechRecognizer.startListening(intent)
   137→        } catch (e: ActivityNotFoundException) {
   138→            isListening = false
   139→        }
   140→    }
   141→
   142→    fun sendTask(text: String) {
   143→        val trimmed = text.trim()
   144→        if (trimmed.isNotBlank()) {
   145→            if (state.goal.isBlank()) {
   146→                viewModel.startTask(trimmed)
   147→            } else {
   148→                viewModel.continueChat(trimmed)
   149→            }
   150→            taskInput = ""
   151→        }
   152→    }
   153→
   154→    Box(
   155→        modifier = Modifier
   156→            .fillMaxSize()
   157→            .background(AppColors.DarkBackground)
   158→    ) {
   159→        Column(modifier = Modifier.fillMaxSize()) {
   160→            // ── Top Bar ──
   161→            TopBar(
   162→                onNewChat = { viewModel.newChat() },
   163→                onHistory = onNavigateToHistory,
   164→                onSettings = onNavigateToSettings,
   165→                onDebug = onNavigateToDebug
   166→            )
   167→
   168→            // ── Status bar (when running) ──
   169→            if (isRunning) {
   170→                RunningStatusBar(state = state)
   171→            }
   172→
   173→            // ── Accessibility banner ──
   174→            if (!isAccessibilityEnabled) {
   175→                AccessibilityBanner(onEnable = { viewModel.openAccessibilitySettings() })
   176→            }
   177→
   178→            // ── Result banners ──
   179→            if (state.status == AgentStatus.COMPLETED && !isRunning) {
   180→                ResultBanner(text = "Done", isSuccess = true)
   181→            } else if (state.status == AgentStatus.FAILED && !isRunning) {
   182→                ResultBanner(text = state.lastError ?: "Something went wrong", isSuccess = false)
   183→            } else if (state.status == AgentStatus.CANCELLED && !isRunning) {
   184→                ResultBanner(text = "Stopped", isSuccess = false)
   185→            }
   186→
   187→            // ── Chat / Event List ──
   188→            if (displayItems.isEmpty() && !isRunning) {
   189→                EmptyState(onSuggestionClick = { sendTask(it) })
   190→            } else {
   191→                LazyColumn(
   192→                    state = listState,
   193→                    modifier = Modifier
   194→                        .fillMaxSize()
   195→                        .padding(horizontal = 16.dp),
   196→                    verticalArrangement = Arrangement.spacedBy(6.dp),
   197→                    contentPadding = PaddingValues(vertical = 8.dp)
   198→                ) {
   199→                    items(
   200→                        items = displayItems,
   201→                        key = { it.timestamp.toString() + it::class.simpleName }
   202→                    ) { item ->
   203→                        when (item) {
   204→                            is DisplayItem.UserMessage -> UserChatBubble(text = item.text)
   205→                            is DisplayItem.AgentMessage -> AgentChatBubble(text = item.text)
   206→                            is DisplayItem.ToolAction -> ToolActionCard(item)
   207→                            is DisplayItem.ErrorItem -> ErrorText(item.message)
   208→                        }
   209→                    }
   210→                    // Bottom padding for input bar
   211→                    item {
   212→                        Spacer(modifier = Modifier.height(80.dp))
   213→                    }
   214→                }
   215→            }
   216→        }
   217→
   218→        // ── Bottom Input Bar ──
   219→        BottomInputBar(
   220→            input = taskInput,
   221→            onInputChange = { taskInput = it },
   222→            onSend = { sendTask(taskInput) },
   223→            onStop = { viewModel.stopAgent() },
   224→            isRunning = isRunning,
   225→            isListening = isListening,
   226→            onVoiceClick = { startVoiceInput() },
   227→            canSend = taskInput.isNotBlank() && !isRunning
   228→        )
   229→
   230→        // ── Dialogs ──
   231→        state.pendingConfirmation?.let { confirmation ->
   232→            ConfirmationDialog(
   233→                confirmation = confirmation,
   234→                onConfirm = { viewModel.respondToConfirmation(true) },
   235→                onDeny = { viewModel.respondToConfirmation(false) }
   236→            )
   237→        }
   238→
   239→        state.pendingQuestion?.let { question ->
   240→            UserQuestionDialog(
   241→                question = question,
   242→                onAnswer = { answer -> viewModel.respondToUser(answer) },
   243→                onDismiss = {}
   244→            )
   245→        }
   246→    }
   247→}

   338→@Composable
   339→private fun RunningStatusBar(state: com.androidagent.aiagent.agent.AgentState) {
   340→    val (label, icon, color) = when (state.status) {
   341→        AgentStatus.THINKING -> Triple("Analyzing screen", Icons.Default.Psychology, AppColors.Info)
   342→        AgentStatus.EXECUTING -> Triple("Using mobile...", Icons.Default.TouchApp, AppColors.Success)
   343→        AgentStatus.WAITING_FOR_USER -> Triple("Waiting for you", Icons.Default.QuestionAnswer, AppColors.Primary)
   344→        AgentStatus.WAITING_FOR_CONFIRMATION -> Triple("Confirm action", Icons.Default.Security, AppColors.Warning)
   345→        AgentStatus.VERIFYING -> Triple("Verifying...", Icons.Default.VerifiedUser, AppColors.Info)
   346→        else -> Triple("Working...", Icons.Default.Autorenew, AppColors.Secondary)
   347→    }
   348→
   349→    val infiniteTransition = rememberInfiniteTransition(label = "pulse")
   350→    val alpha by infiniteTransition.animateFloat(
   351→        initialValue = 0.3f, targetValue = 1f,
   352→        animationSpec = infiniteRepeatable(tween(800, easing = LinearEasing), RepeatMode.Reverse),
   353→        label = "alpha"
   354→    )
   355→
   356→    Row(
   357→        modifier = Modifier
   358→            .fillMaxWidth()
   359→            .background(color.copy(alpha = 0.08f))
   360→            .padding(horizontal = 16.dp, vertical = 8.dp),
   361→        verticalAlignment = Alignment.CenterVertically
   362→    ) {
   363→        Icon(icon, contentDescription = null, tint = color, modifier = Modifier.size(16.dp).alpha(alpha))
   364→        Spacer(modifier = Modifier.width(10.dp))
   365→        Text(label, color = color, fontSize = 13.sp, fontWeight = FontWeight.Medium, modifier = Modifier.weight(1f))
   366→        Text("${state.durationFormatted}", color = AppColors.TextMuted, fontSize = 11.sp)
   367→        Spacer(modifier = Modifier.width(12.dp))
   368→        Text("Step ${state.stepNumber}", color = AppColors.TextMuted, fontSize = 11.sp)
   369→    }
   370→}

   457→        verticalAlignment = Alignment.CenterVertically
   458→    ) {
   459→        Icon(icon, contentDescription = null, tint = color, modifier = Modifier.size(14.dp))
   460→        Spacer(modifier = Modifier.width(8.dp))

   462→            desc,
   463→            color = color,
   464→            fontSize = 12.sp,
   465→            maxLines = 1,
   466→            modifier = Modifier.weight(1f)
   467→        )
   468→        if (action.success) {
   469→            Icon(Icons.Default.Check, contentDescription = null, tint = AppColors.Success, modifier = Modifier.size(14.dp))
   470→        } else {
   471→            Icon(Icons.Default.Close, contentDescription = null, tint = AppColors.Error, modifier = Modifier.size(14.dp))

   473→    }

   735→private fun describeAction(toolName: String, arguments: String): String {
   736→    val text = extractJsonField(arguments, "text")
   737→    val appName = extractJsonField(arguments, "app_name")
   738→    val direction = extractJsonField(arguments, "direction")
   739→    return when {
   740→        toolName.contains("launch_app") -> if (!appName.isNullOrBlank()) "Opened $appName" else "Opened app"
   741→        toolName.contains("click") -> "Tapped element"
   742→        toolName.contains("long_click") -> "Long pressed"
   743→        toolName.contains("type_text") -> if (!text.isNullOrBlank()) "Typed: ${text.take(40)}" else "Typed text"
   744→        toolName.contains("clear_text") -> "Cleared text"
   745→        toolName.contains("scroll") -> "Scrolled ${direction ?: ""}"
   746→        toolName.contains("swipe") -> {
   747→            val d = extractJsonField(arguments, "direction") ?: ""
   748→            when {
   749→                d.contains("up", ignoreCase = true) -> "Swiped up"
   750→                d.contains("down", ignoreCase = true) -> "Swiped down"
   751→                d.contains("left", ignoreCase = true) -> "Swiped left"
   752→                d.contains("right", ignoreCase = true) -> "Swiped right"
   753→                else -> "Swiped"
   754→            }
   755→        }
   756→        toolName.contains("back") -> "Went back"
   757→        toolName.contains("home") -> "Went home"
   758→        toolName.contains("recents") -> "Opened recents"
   759→        toolName.contains("press_key") -> "Pressed key"
   760→        toolName.contains("wait") -> "Waiting..."
   761→        toolName.contains("screenshot") -> "Took screenshot"
   762→        toolName.contains("inspect") -> "Read screen"
   763→        toolName.contains("find") -> "Searching..."
   764→        toolName.contains("analyze") -> "Analyzing screen"
   765→        toolName.contains("visual") -> "Looking for element"
   766→        toolName.contains("finish") -> "Done"
   767→        toolName.contains("stop") -> "Stopped"
   768→        else -> toolName.substringAfterLast(".").replace("_", " ")
   769→    }
   770→}

   773→    toolName.contains("launch") -> Icons.Default.Launch

   789→private fun extractJsonField(json: String, field: String): String? {
   790→    return Regex(""""$field"\s*:\s*"([^"\x27]*)"""", RegexOption.IGNORE_CASE).find(json)?.groupValues?.getOrNull(1)
