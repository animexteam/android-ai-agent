    21→    21→/**
    22→    22→ * The heart of the agent – runs the observe → reason → act → verify loop.
    23→    23→ *
    24→    24→ * Architecture:
    25→    25→ * - The runtime owns a single [MutableStateFlow] of [AgentState] that the
    26→    26→ *   UI observes.  All state mutations go through `copy()` / `withEvent()`.
    27→    27→ * - The main loop ([runAgentLoop]) is launched in a dedicated
    28→    28→ *   [SupervisorJob] + [Dispatchers.Main] scope so that the UI stays
    29→    29→ *   responsive.
    30→    30→ * - Prompt construction is **entirely delegated** to [AgentPromptBuilder].
    31→    31→ * - Tool name resolution (aliases, fuzzy match) is handled by
    32→    32→ *   [ToolExecutor.resolveToolName].
    33→    33→ * - Loop detection is delegated to [AgentLoopGuard].
    34→    34→ */
    35→    35→class AgentRuntime(
    36→    36→    private val gemmaClient: GemmaClient,
    37→    37→    private val toolRegistry: ToolRegistry,
    38→    38→    private val toolExecutor: ToolExecutor,
    39→    39→    private val accessibilityObserver: AccessibilityObserver,
    40→    40→    private val settingsRepository: SettingsRepository,
    41→    41→    private val taskRepository: TaskRepository? = null,
    42→    42→    private val promptBuilder: AgentPromptBuilder = AgentPromptBuilder(),
    43→    43→    private val loopGuard: AgentLoopGuard = AgentLoopGuard()
    44→    44→) {
    45→    45→
    46→    46→    companion object {
    47→    47→        private const val TAG = "AgentRuntime"
    48→    48→        /** Maximum model call retries on transient errors. */
    49→    49→        private const val MAX_MODEL_RETRIES = 2
    50→    50→    }
    51→    51→

    53→    53→    // State

    55→    55→
    56→    56→    private val _state = MutableStateFlow(AgentState())
    57→    57→    val state: StateFlow<AgentState> = _state.asStateFlow()
    58→    58→
    59→    59→    private var agentJob: Job? = null
    60→    60→    private val agentScope = CoroutineScope(SupervisorJob() + Dispatchers.Main)
    61→    61→
    62→    62→    /**
    63→    63→     * System prompt is built once per task (not every turn) because it
    64→    64→     * contains the tool catalogue which doesn't change during a task.
    65→    65→     */
    66→    66→    private var cachedSystemPrompt: String? = null
    67→    67→
    68→    68→    /**
    69→    69→     * Known tool names, refreshed when the system prompt is built.
    70→    70→     * Passed to [DecisionParser] so it can validate tool names.
    71→    71→     */
    72→    72→    private var knownToolNames: Set<String> = emptySet()
    73→    73→

    78→    78→    /** Start a new task, cancelling any running task first. */
    79→    79→    fun startTask(goal: String) {
    80→    80→        agentJob?.cancel()
    81→    81→        loopGuard.reset()
    82→    82→        cachedSystemPrompt = null
    83→    83→
    84→    84→        _state.value = AgentState(
    85→    85→            goal = goal,
    86→    86→            status = AgentStatus.THINKING,
    87→    87→            stepNumber = 0,
    88→    88→            startTime = System.currentTimeMillis()
    89→    89→        )
    90→    90→
    91→    91→        agentJob = agentScope.launch {
    92→    92→            runAgentLoop()
    93→    93→        }
    94→    94→    }
    95→    95→
    96→    96→    /** Resume the agent after the user answered a question. */
    97→    97→    fun respondToUser(answer: String) {
    98→    98→        val currentState = _state.value
    99→    99→        if (currentState.status != AgentStatus.WAITING_FOR_USER) return
   100→   100→
   101→   101→        val userEvent = AgentEvent.UserMessage(
   102→   102→            stepNumber = currentState.stepNumber,
   103→   103→            text = answer
   104→   104→        )
   105→   105→
   106→   106→        _state.value = currentState
   107→   107→            .withEvent(userEvent)
   108→   108→            .copy(
   109→   109→                pendingQuestion = null,
   110→   110→                status = AgentStatus.THINKING
   111→   111→            )
   112→   112→
   113→   113→        agentJob = agentScope.launch { runAgentLoop() }
   114→   114→    }
   115→   115→
   116→   116→    /** Resume the agent after the user responded to a safety confirmation. */
   117→   117→    fun respondToConfirmation(confirmed: Boolean) {
   118→   118→        val currentState = _state.value
   119→   119→        if (currentState.status != AgentStatus.WAITING_FOR_CONFIRMATION) return
   120→   120→        val pending = currentState.pendingConfirmation ?: return
   121→   121→
   122→   122→        if (confirmed) {
   123→   123→            agentJob = agentScope.launch {
   124→   124→                try {
   125→   125→                    val observationId = currentState.currentObservationId
   126→   126→                    val argsJson = try {
   127→   127→                        kotlinx.serialization.json.Json.parseToJsonElement(pending.arguments)
   128→   128→                    } catch (_: Exception) { null }
   129→   129→                    val args = argsJson as? kotlinx.serialization.json.JsonObject
   130→   130→                        ?: kotlinx.serialization.json.JsonObject(emptyMap())
   131→   131→
   132→   132→                    val toolResult = withContext(Dispatchers.IO) {
   133→   133→                        toolExecutor.execute(pending.toolName, args, observationId)
   134→   134→                    }
   135→   135→
   136→   136→                    val toolEvent = AgentEvent.ToolExecution(
   137→   137→                        stepNumber = currentState.stepNumber,
   138→   138→                        toolName = pending.toolName,
   139→   139→                        arguments = pending.arguments,
   140→   140→                        result = toolResult
   141→   141→                    )
   142→   142→
   143→   143→                    loopGuard.recordAction(pending.toolName, pending.arguments)
   144→   144→
   145→   145→                    _state.value = _state.value
   146→   146→                        .withEvent(toolEvent)
   147→   147→                        .copy(
   148→   148→                            pendingConfirmation = null,
   149→   149→                            status = AgentStatus.THINKING,
   150→   150→                            stepNumber = _state.value.stepNumber + 1
   151→   151→                        )
   152→   152→
   153→   153→                    runAgentLoop()
   154→   154→                } catch (e: CancellationException) {
   155→   155→                    throw e
   156→   156→                } catch (e: Exception) {
   157→   157→                    Log.e(TAG, "Failed to execute confirmed action", e)
   158→   158→                    _state.value = _state.value.copy(
   159→   159→                        pendingConfirmation = null,
   160→   160→                        status = AgentStatus.THINKING,
   161→   161→                        lastError = e.message
   162→   162→                    )
   163→   163→                    runAgentLoop()
   164→   164→                }
   165→   165→            }
   166→   166→        } else {
   167→   167→            val errorEvent = AgentEvent.Error(
   168→   168→                stepNumber = currentState.stepNumber,
   169→   169→                message = "User rejected the action: ${pending.toolName}"
   170→   170→            )
   171→   171→            _state.value = currentState
   172→   172→                .withEvent(errorEvent)
   173→   173→                .copy(
   174→   174→                    pendingConfirmation = null,
   175→   175→                    status = AgentStatus.THINKING
   176→   176→                )
   177→   177→
   178→   178→            agentJob = agentScope.launch { runAgentLoop() }
   179→   179→        }
   180→   180→    }
   181→   181→
   182→   182→    /** Cancel the current task. */
   183→   183→    fun stopAgent() {
   184→   184→        Log.i(TAG, "Stop requested, cancelling agent job")
   185→   185→        agentJob?.cancel()
   186→   186→        agentJob = null
   187→   187→        _state.value = _state.value.copy(
   188→   188→            status = AgentStatus.CANCELLED,
   189→   189→            endTime = System.currentTimeMillis(),
   190→   190→            pendingConfirmation = null,
   191→   191→            pendingQuestion = null
   192→   192→        )
   193→   193→    }
   194→   194→
   195→   195→    // ===================================================================
   196→   196→    // Main agent loop
   197→   197→    // ===================================================================
   198→   198→
   199→   199→    private suspend fun runAgentLoop() {
   200→   200→        try {
   201→   201→            val maxSteps = withContext(Dispatchers.IO) {
   202→   202→                settingsRepository.maxSteps()
   203→   203→            }
   204→   204→            _state.value = _state.value.copy(maxSteps = maxSteps)
   205→   205→
   206→   206→            // Build system prompt once (includes current tool catalogue)
   207→   207→            ensureSystemPrompt()
   208→   208→
   209→   209→            while (_state.value.status.isActive) {
   210→   210→                val current = _state.value
   211→   211→
   212→   212→                // --- Step budget check ---
   213→   213→                if (current.stepNumber >= maxSteps) {
   214→   214→                    finishTask(
   215→   215→                        status = AgentStatus.FAILED,
   216→   216→                        error = "Max steps ($maxSteps) reached without completing the task."
   217→   217→                    )
   218→   218→                    break
   219→   219→                }
   220→   220→
   221→   221→                // --- Observe ---
   222→   222→                val observation = observeScreen()
   223→   223→                if (observation == null) continue  // error already recorded
   224→   224→
   225→   225→                // --- Loop detection ---
   226→   226→                val loopWarning = if (loopGuard.isLooping()) {
   227→   227→                    loopGuard.getLoopMessage()
   228→   228→                } else null
   229→   229→
   230→   230→                // --- Call model ---
   231→   231→                val decision = callModel(observation, loopWarning)
   232→   232→                    ?: continue  // model error already recorded, step consumed
   233→   233→
   234→   234→                // --- Record model event ---
   235→   235→                val modelEvent = AgentEvent.ModelResponse(
   236→   236→                    stepNumber = _state.value.stepNumber,
   237→   237→                    decisionType = decision.type,
   238→   238→                    content = describeDecision(decision)
   239→   239→                )
   240→   240→                _state.value = _state.value.withEvent(modelEvent)
   241→   241→
   242→   242→                // --- Execute decision ---
   243→   243→                val shouldContinue = executeDecision(decision, observation.id)
   244→   244→                if (!shouldContinue) break
   245→   245→
   246→   246→                // --- Advance step ---
   247→   247→                _state.value = _state.value.copy(
   248→   248→                    stepNumber = _state.value.stepNumber + 1
   249→   249→                )
   250→   250→            }
   251→   251→        } catch (e: CancellationException) {
   252→   252→            Log.i(TAG, "Agent loop cancelled")
   253→   253→        } catch (e: Exception) {
   254→   254→            Log.e(TAG, "Agent loop error", e)
   255→   255→            finishTask(
   256→   256→                status = AgentStatus.FAILED,
   257→   257→                error = e.message ?: "Unknown error"
   258→   258→            )
   259→   259→        }
   260→   260→    }
   261→   261→
   262→   262→    // ===================================================================
   263→   263→    // Step 1: Observe
   264→   264→    // ===================================================================
   265→   265→
   266→   266→    /**
   267→   267→     * Observe the screen and update state.  Returns the observation,
   268→   268→     * or null if observation failed (in which case an error event is
   269→   269→     * recorded and the step is consumed).
   270→   270→     */
   271→   271→    private suspend fun observeScreen(): AndroidObservation? {
   272→   272→        _state.value = _state.value.copy(status = AgentStatus.THINKING)
   273→   273→
   274→   274→        val useVision = shouldUseVision()
   275→   275→        val observation: AndroidObservation
   276→   276→        try {
   277→   277→            observation = withContext(Dispatchers.IO) {
   278→   278→                accessibilityObserver.observeWithScreenshot(takeScreenshot = useVision)
   279→   279→            }
   280→   280→        } catch (e: CancellationException) {
   281→   281→            throw e
   282→   282→        } catch (e: Exception) {
   283→   283→            Log.w(TAG, "Observation failed: ${e.message}")
   284→   284→            _state.value = _state.value
   285→   285→                .withEvent(
   286→   286→                    AgentEvent.Error(
   287→   287→                        stepNumber = _state.value.stepNumber,
   288→   288→                        message = "Failed to observe screen: ${e.message}"
   289→   289→                    )
   290→   290→                )
   291→   291→                .copy(stepNumber = _state.value.stepNumber + 1)
   292→   292→            return null
   293→   293→        }
   294→   294→
   295→   295→        val obsEvent = AgentEvent.Observation(
   296→   296→            stepNumber = _state.value.stepNumber,
   297→   297→            summary = "Package: ${observation.packageName}, " +
   298→   298→                "Activity: ${observation.activityName}, " +
   299→   299→                "Nodes: ${observation.uiTree.size}, " +
   300→   300→                "Has screenshot: ${observation.screenshotBase64 != null}"
   301→   301→        )
   302→   302→
   303→   303→        loopGuard.recordObservation(observation)
   304→   304→
   305→   305→        _state.value = _state.value
   306→   306→            .withEvent(obsEvent)
   307→   307→            .copy(
   308→   308→                lastObservation = observation,
   309→   309→                currentPackage = observation.packageName,
   310→   310→                currentObservationId = observation.id
   311→   311→            )
   312→   312→
   313→   313→        return observation
   314→   314→    }
   315→   315→
   316→   316→    // ===================================================================
   317→   317→    // Step 2: Call model
   318→   318→    // ===================================================================
   319→   319→
   320→   320→    /**
   321→   321→     * Call the AI model with the current screen state and history.
   322→   322→     * Returns the parsed decision, or null if the call failed (error
   323→   323→     * is recorded and step consumed).
   324→   324→     */
   325→   325→    private suspend fun callModel(
   326→   326→        observation: AndroidObservation,
   327→   327→        loopWarning: String?
   328→   328→       ): AgentDecision? {
   329→   329→        val current = _state.value
   330→   330→        val systemPrompt = cachedSystemPrompt!!
   331→   331→
   332→   332→        val userMessage = promptBuilder.buildUserMessage(
   333→   333→            goal = current.goal,
   334→   334→            observation = observation,
   335→   335→            history = current.history,
   336→   336→            loopWarning = loopWarning
   337→   337→        )
   338→   338→
   339→   339→        val screenshotBase64 = observation.screenshotBase64
   340→   340→        var lastError: String? = null
   341→   341→
   342→   342→        // Retry transient model failures up to MAX_MODEL_RETRIES times.
   343→   343→        for (attempt in 0..MAX_MODEL_RETRIES) {
   344→   344→            val callStartTime = System.currentTimeMillis()
   345→   345→            try {
   346→   346→                val response = withContext(Dispatchers.IO) {
   347→   347→                    gemmaClient.generate(
   348→   348→                        systemPrompt = systemPrompt,
   349→   349→                        userMessage = userMessage,
   350→   350→                        tools = toolRegistry.getAll(),
   351→   351→                        screenshotBase64 = screenshotBase64
   352→   352→                    )
   353→   353→                }
   354→   354→
   355→   355→                _state.value = _state.value.copy(
   356→   356→                    modelLatencyMs = System.currentTimeMillis() - callStartTime
   357→   357→                )
   358→   358→
   359→   359→                return DecisionParser.parse(response, knownToolNames)
   360→   360→
   361→   361→            } catch (e: CancellationException) {
   362→   362→                throw e
   363→   363→            } catch (e: Exception) {
   364→   364→                lastError = e.message
   365→   365→                Log.w(TAG, "Model call failed (attempt ${attempt + 1}/$MAX_MODEL_RETRIES): ${e.message}")
   366→   366→                if (attempt == MAX_MODEL_RETRIES) break
   367→   367→                // Brief pause before retry
   368→   368→                kotlinx.coroutines.delay(500L * (attempt + 1))
   369→   369→            }
   370→   370→        }
   371→   371→
   372→   372→        // All retries exhausted
   373→   373→        Log.w(TAG, "Model call failed after all retries: $lastError")
   374→   374→        _state.value = _state.value
   375→   375→            .withEvent(
   376→   376→                AgentEvent.Error(
   377→   377→                    stepNumber = _state.value.stepNumber,
   378→   378→                    message = "Model call failed: $lastError"
   379→   379→                )
   380→   380→            )
   381→   381→            .copy(
   382→   382→                lastError = "Model error: $lastError",
   383→   383→                stepNumber = _state.value.stepNumber + 1
   384→   384→            )
   385→   385→        return null
   386→   386→    }
   387→   387→
   388→   388→    // ===================================================================
   389→   389→    // Step 3: Execute decision
   390→   390→    // ===================================================================
   391→   391→
   392→   392→    /**
   393→   393→     * Execute the parsed model decision.  Returns `true` if the loop
   394→   394→     * should continue, `false` if the task is done or paused.
   395→   395→     */
   396→   396→    private suspend fun executeDecision(
   397→   397→        decision: AgentDecision,
   398→   398→        observationId: String
   399→   399→    ): Boolean {
   400→   400→        return when (decision) {
   401→   401→            is ToolCallDecision -> executeToolCall(decision, observationId)
   402→   402→            is AskUserDecision -> {
   403→   403→                _state.value = _state.value.copy(
   404→   404→                    status = AgentStatus.WAITING_FOR_USER,
   405→   405→                    pendingQuestion = decision.question
   406→   406→                )
   407→   407→                false  // pause loop
   408→   408→            }
   409→   409→            is FinishDecisionData -> {
   410→   410→                val newStatus = if (decision.success) AgentStatus.COMPLETED else AgentStatus.FAILED
   411→   411→                finishTask(
   412→   412→                    status = newStatus,
   413→   413→                    error = if (!decision.success) decision.message else null
   414→   414→                )
   415→   415→                false  // stop loop
   416→   416→            }
   417→   417→            is MessageDecision -> {
   418→   418→                // Status message – nothing to do, loop continues.
   419→   419→                true
   420→   420→            }
   421→   421→            is ErrorDecisionData -> {
   422→   422→                _state.value = _state.value
   423→   423→                    .withEvent(
   424→   424→                        AgentEvent.Error(
   425→   425→                            stepNumber = _state.value.stepNumber,
   426→   426→                            message = decision.message
   427→   427→                        )
   428→   428→                    )
   429→   429→                    .copy(lastError = decision.message)
   430→   430→                true  // loop continues so the model can recover
   431→   431→            }
   432→   432→        }
   433→   433→    }
   434→   434→
   435→   435→    // -------------------------------------------------------------------
   436→   436→    // Tool call execution
   437→   437→    // -------------------------------------------------------------------
   438→   438→
   439→   439→    private suspend fun executeToolCall(
   440→   440→        decision: ToolCallDecision,
   441→   441→        observationId: String
   442→   442→    ): Boolean {
   443→   443→        _state.value = _state.value.copy(status = AgentStatus.EXECUTING)
   444→   444→
   445→   445→        // Resolve aliases before execution
   446→   446→        val resolution = toolExecutor.resolveToolName(decision.toolName)
   447→   447→        val toolName = resolution.resolvedName
   448→   448→
   449→   449→        // If the model used a wrong name, log it but execute the corrected name.
   450→   450→        if (resolution.wasCorrected) {
   451→   451→            Log.i(TAG, "Auto-corrected tool name: ${decision.toolName} → $toolName (${resolution.method})")
   452→   452→        }
   453→   453→
   454→   454→        // Execute (safety checks are inside ToolExecutor)
   455→   455→        var toolResult: ToolResult
   456→   456→        try {
   457→   457→            toolResult = withContext(Dispatchers.IO) {
   458→   458→                toolExecutor.execute(toolName, decision.arguments, observationId)
   459→   459→            }
   460→   460→        } catch (e: CancellationException) {
   461→   461→            throw e
   462→   462→        } catch (e: Exception) {
   463→   463→            Log.e(TAG, "Tool execution exception for $toolName", e)
   464→   464→            _state.value = _state.value
   465→   465→                .withEvent(
   466→   466→                    AgentEvent.Error(
   467→   467→                        stepNumber = _state.value.stepNumber,
   468→   468→                        message = "Tool execution exception for $toolName: ${e.message}"
   469→   469→                    )
   470→   470→                    )
   471→   471→                .copy(status = AgentStatus.THINKING, lastError = e.message)
   472→   472→            return true
   473→   473→        }
   474→   474→
   475→   475→        // Handle UNKNOWN_TOOL (should be rare since we resolve aliases first)
   476→   476→        if (!toolResult.success && toolResult.error?.code == ToolExecutor.ERROR_UNKNOWN_TOOL) {
   477→   477→            _state.value = _state.value
   478→   478→                .withEvent(
   479→   479→                    AgentEvent.Error(
   480→   480→                        stepNumber = _state.value.stepNumber,
   481→   481→                        message = buildString {
   482→   482→                            append("Tool '$toolName' does not exist. ")
   483→   483→                            append("Available: ${toolExecutor.getRegisteredHandlerNames().sorted().joinToString(", ")}.")
   484→   484→                            append(" Please use a valid tool name.")
   485→   485→                        }
   486→   486→                    )
   487→   487→                )
   488→   488→                .copy(status = AgentStatus.THINKING, lastError = "Unknown tool: $toolName")
   489→   489→            return true
   490→   490→        }
   491→   491→
   492→   492→        // Handle NODE_NOT_FOUND with automatic re-observe + retry
   493→   493→        if (!toolResult.success && toolResult.error?.code == "NODE_NOT_FOUND") {
   494→   494→            toolResult = retryAfterReobserve(decision, toolResult)
   495→   495→        }
   496→   496→
   497→   497→        // Record the tool execution event
   498→   498→        val toolEvent = AgentEvent.ToolExecution(
   499→   499→            stepNumber = _state.value.stepNumber,
   500→   500→            toolName = toolName,
   501→   501→            arguments = decision.arguments.toString(),
   502→   502→            result = toolResult
   503→   503→        )
   504→   504→        loopGuard.recordAction(toolName, decision.arguments.toString())
   505→   505→        _state.value = _state.value.withEvent(toolEvent)
   506→   506→
   507→   507→        // Handle post-execution status transitions
   508→   508→        return handlePostExecutionStatus(toolResult, decision)
   509→   509→    }
   510→   510→
   511→   511→    /**
   512→   512→     * When a NODE_NOT_FOUND error occurs, re-observe the screen (without
   513→   513→     * screenshot) and retry the same tool call once.
   514→   514→     */
   515→   515→    private suspend fun retryAfterReobserve(
   516→   516→        decision: ToolCallDecision,
   517→   517→        originalResult: ToolResult
   518→   518→    ): ToolResult {
   519→   519→        Log.i(TAG, "NODE_NOT_FOUND for ${decision.toolName}, re-observing and retrying once")
   520→   520→
   521→   521→        return try {
   522→   522→            val newObservation = withContext(Dispatchers.IO) {
   523→   523→                accessibilityObserver.observeWithScreenshot(takeScreenshot = false)
   524→   524→            }
   525→   525→
   526→   526→            val retryObsEvent = AgentEvent.Observation(
   527→   527→                stepNumber = _state.value.stepNumber,
   528→   528→                summary = "Auto re-observe (NODE_NOT_FOUND retry): " +
   529→   529→                    "Package: ${newObservation.packageName}, Nodes: ${newObservation.uiTree.size}"
   530→   530→            )
   531→   531→            loopGuard.recordObservation(newObservation)
   532→   532→
   533→   533→            _state.value = _state.value
   534→   534→                .withEvent(retryObsEvent)
   535→   535→                .copy(
   536→   536→                    lastObservation = newObservation,
   537→   537→                    currentObservationId = newObservation.id
   538→   538→                )
   539→   539→
   540→   540→            withContext(Dispatchers.IO) {
   541→   541→                toolExecutor.execute(decision.toolName, decision.arguments, newObservation.id)
   542→   542→            }
   543→   543→        } catch (e: CancellationException) {
   544→   544→            throw e
   545→   545→        } catch (e: Exception) {
   546→   546→            Log.w(TAG, "Re-observation/retry failed, using original NODE_NOT_FOUND result", e)
   547→   547→            originalResult
   548→   548→        }
   549→   549→    }

   551→   551→    /**
   552→   552→     * After a tool call completes, transition the agent's status based
   553→   553→     * on the result's error code.
   554→   554→     */
   555→   555→    private fun handlePostExecutionStatus(
   556→   556→        toolResult: ToolResult,
   557→   557→        decision: ToolCallDecision
   558→   558→    ): Boolean {
   559→   559→        return when {
   560→   560→            toolResult.error?.code == "NODE_STALE" -> {
   561→   561→                _state.value = _state.value.copy(
   562→   562→                    status = AgentStatus.THINKING,
   563→   563→                    lastError = "Node was stale, will re-observe"
   564→   564→                )
   565→   565→                true
   566→   566→            }
   567→   567→            toolResult.error?.code == ToolExecutor.ERROR_CONFIRMATION_REQUIRED -> {
   568→   568→                _state.value = _state.value.copy(
   569→   569→                    status = AgentStatus.WAITING_FOR_CONFIRMATION,
   570→   570→                    pendingConfirmation = PendingConfirmation(
   571→   571→                        toolName = decision.toolName,
   572→   572→                        arguments = decision.arguments.toString(),
   573→   573→                        reason = toolResult.error.message,
   574→   574→                        stepNumber = _state.value.stepNumber
   575→   575→                    )
   576→   576→                )
   577→   577→                false  // pause loop
   578→   578→            }
   579→   579→            toolResult.error?.code == ToolExecutor.ERROR_BLOCKED -> {
   580→   580→                _state.value = _state.value
   581→   581→                    .withEvent(
   582→   582→                        AgentEvent.Error(
   583→   583→                            stepNumber = _state.value.stepNumber,
   584→   584→                            message = "Action blocked: ${toolResult.error.message}"
   585→   585→                        )
   586→   586→                    )
   587→   587→                    .copy(
   588→   588→                        status = AgentStatus.THINKING,
   589→   589→                        lastError = toolResult.error.message
   590→   590→                    )
   591→   591→                true
   592→   592→            }
   593→   593→            !toolResult.success -> {
   594→   594→                _state.value = _state.value.copy(
   595→   595→                    status = AgentStatus.THINKING,
   596→   596→                    lastError = toolResult.error?.message ?: "Tool failed"
   597→   597→                )
   598→   598→                true
   599→   599→            }
   600→   600→            else -> {
   601→   601→                _state.value = _state.value.copy(status = AgentStatus.THINKING)
   602→   602→                true
   603→   603→            }
   604→   604→        }
   605→   605→    }

   624→   624→    /** Determine whether to include a screenshot based on vision mode settings. */
   625→   625→    private suspend fun shouldUseVision(): Boolean {
   626→   626→        return withContext(Dispatchers.IO) {
   627→   627→            when (settingsRepository.visionMode()) {
   628→   628→                "ALWAYS" -> true
   629→   629→                "OFF" -> false
   630→   630→                "AUTO" -> {
   631→   631→                    val lastObs = _state.value.lastObservation
   632→   632→                    val actionableCount = lastObs?.uiTree
   633→   633→                        ?.count { it.isClickable || it.isEditable } ?: 0
   634→   634→                    val lastToolFailed = _state.value.history
   635→   635→                        .filterIsInstance<AgentEvent.ToolExecution>()
   636→   636→                        .lastOrNull()
   637→   637→                        ?.let { !it.result.success } ?: false
   638→   638→                    actionableCount < 3 || lastToolFailed
   639→   639→                }
   640→   640→                else -> false
   641→   641→            }
   642→   642→        }
   643→   643→    }

   656→   656→    /** Transition the agent to a terminal state and persist the task. */
   657→   657→    private fun finishTask(status: AgentStatus, error: String?) {
   658→   658→        _state.value = _state.value.copy(
   659→   659→            status = status,
   660→   660→            endTime = System.currentTimeMillis(),

   664→   664→    }

   666→   666→    /** Persist the completed/failed task to the database (fire-and-forget). */
   667→   667→    private fun saveTask() {
   668→   668→        val currentState = _state.value
   669→   669→        taskRepository?.let { repo ->
   670→   670→            agentScope.launch {
   671→   671→                try {
   672→   672→                    withContext(Dispatchers.IO) {
   673→   673→                        val record = com.androidagent.aiagent.data.TaskRecord(
   674→   674→                            goal = currentState.goal,
   675→   675→                            status = currentState.status.name,
   676→   676→                            startTime = currentState.startTime ?: System.currentTimeMillis(),
   677→   677→                            endTime = currentState.endTime,
   678→   678→                            stepCount = currentState.stepNumber,
   679→   679→                            result = if (currentState.status == AgentStatus.COMPLETED)
   680→   680→                                "Completed" else currentState.lastError
   681→   681→                        )
   682→   682→                        repo.insertTask(record)
   683→   683→                    }
   684→   684→                } catch (_: Exception) { /* best-effort persistence */ }
   685→   685→            }
   686→   686→        }
   687→   687→    }
