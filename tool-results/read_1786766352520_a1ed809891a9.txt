    43→    43→@OptIn(ExperimentalMaterial3Api::class)
    44→    44→@Composable
    45→    45→fun MainScreen(
    46→    46→    viewModel: AgentViewModel,
    47→    47→    onNavigateToSettings: () -> Unit,
    48→    48→    onNavigateToDebug: () -> Unit,
    49→    49→    onNavigateToHistory: () -> Unit
    50→    50→) {
    51→    51→    val state by viewModel.stateFlow.collectAsState()
    52→    52→    val isAccessibilityEnabled by remember {
    53→    53→        mutableStateOf(viewModel.isAccessibilityServiceEnabled())
    54→    54→    }
    55→    55→    var taskInput by remember { mutableStateOf("") }
    56→    56→    val listState = rememberLazyListState()
    57→    57→    val isRunning = state.status.isActive
    58→    58→    val context = LocalContext.current
    59→    59→
    60→    60→    // Voice input state
    61→    61→    var isListening by remember { mutableStateOf(false) }
    62→    62→    var voiceText by remember { mutableStateOf("") }
    63→    63→
    64→    64→    // Build display items: chat messages + tool actions
    65→    65→    val displayItems = remember(state.history) {
    66→    66→        state.history.mapNotNull { event ->
    67→    67→            when (event) {
    68→    68→                is AgentEvent.UserMessage -> DisplayItem.UserMessage(event.text, event.timestamp)
    69→    69→                is AgentEvent.AgentMessage -> DisplayItem.AgentMessage(event.text, event.timestamp)
    70→    70→                is AgentEvent.ToolExecution -> DisplayItem.ToolAction(
    71→    71→                    name = event.toolName,
    72→    72→                    args = event.arguments,
    73→    73→                    success = event.result.success,
    74→    74→                    error = event.result.error?.message
    75→    75→                )
    76→    76→                is AgentEvent.Error -> DisplayItem.ErrorItem(event.message)
    77→    77→                else -> null
    78→    78→            }
    79→    79→        }
    80→    80→    }
    81→    81→
    82→    82→    LaunchedEffect(displayItems.size) {
    83→    83→        if (displayItems.isNotEmpty()) {
    84→    84→            listState.animateScrollToItem(displayItems.size - 1)
    85→    85→        }
    86→    86→    }
    87→    87→
    88→    88→    // Voice recognizer
    89→    89→    val speechRecognizer = remember {
    90→    90→        try {
    91→    91→            SpeechRecognizer.createSpeechRecognizer(context)
    92→    92→        } catch (e: Exception) { null }
    93→    93→    }
    94→    94→
    95→    95→    DisposableEffect(Unit) {
    96→    96→        onDispose {
    97→    97→            speechRecognizer?.destroy()
    98→    98→        }
    99→    99→    }
   100→   100→
   101→   101→    fun startVoiceInput() {
   102→   102→        if (speechRecognizer == null) return
   103→   103→        isListening = true
   104→   104→        voiceText = ""
   105→   105→        val intent = Intent(RecognizerIntent.ACTION_RECOGNIZE_SPEECH).apply {
   106→   106→            putExtra(RecognizerIntent.EXTRA_LANGUAGE_MODEL, RecognizerIntent.LANGUAGE_MODEL_FREE_FORM)
   107→   107→            putExtra(RecognizerIntent.EXTRA_LANGUAGE, "en-IN")
   108→   108→            putExtra(RecognizerIntent.EXTRA_PROMPT, "Speak to Android-Use...")
   109→   109→        }
   110→   110→        try {
   111→   111→            speechRecognizer.setRecognitionListener(object : RecognitionListener {
   112→   112→                override fun onReadyForSpeech(params: Bundle?) {}
   113→   113→                override fun onBeginningOfSpeech() {}
   114→   114→                override fun onRmsChanged(rmsdB: Float) {}
   115→   115→                override fun onBufferReceived(buffer: ByteArray?) {}
   116→   116→                override fun onEndOfSpeech() {
   117→   117→                    isListening = false
   118→   118→                }
   119→   119→                override fun onError(error: Int) {
   120→   120→                    isListening = false
   121→   121→                }
   122→   122→                override fun onResults(results: Bundle?) {
   123→   123→                    val matches = results?.getStringArrayList(SpeechRecognizer.RESULTS_RECOGNITION)
   124→   124→                    val text = matches?.firstOrNull() ?: ""
   125→   125→                    if (text.isNotBlank()) {
   126→   126→                        taskInput = text
   127→   127→                    }
   128→   128→                    isListening = false
   129→   129→                }
   130→   130→                override fun onPartialResults(partialResults: Bundle?) {
   131→   131→                    val matches = partialResults?.getStringArrayList(SpeechRecognizer.RESULTS_RECOGNITION)
   132→   132→                    voiceText = matches?.firstOrNull() ?: ""
   133→   133→                }
   134→   134→                override fun onEvent(eventType: Int, params: Bundle?) {}
   135→   135→            })
   136→   136→            speechRecognizer.startListening(intent)
   137→   137→        } catch (e: ActivityNotFoundException) {
   138→   138→            isListening = false
   139→   139→        }
   140→   140→    }
   141→   141→
   142→   142→    fun sendTask(text: String) {
   143→   143→        val trimmed = text.trim()
   144→   144→        if (trimmed.isNotBlank()) {
   145→   145→            if (state.goal.isBlank()) {
   146→   146→                viewModel.startTask(trimmed)
   147→   147→            } else {
   148→   148→                viewModel.continueChat(trimmed)
   149→   149→            }
   150→   150→            taskInput = ""
   151→   151→        }
   152→   152→    }
   153→   153→
   154→   154→    Box(
   155→   155→        modifier = Modifier
   156→   156→            .fillMaxSize()
   157→   157→            .background(AppColors.DarkBackground)
   158→   158→    ) {
   159→   159→        Column(modifier = Modifier.fillMaxSize()) {
   160→   160→            // ── Top Bar ──
   161→   161→            TopBar(
   162→   162→                onNewChat = { viewModel.newChat() },
   163→   163→                onHistory = onNavigateToHistory,
   164→   164→                onSettings = onNavigateToSettings,
   165→   165→                onDebug = onNavigateToDebug
   166→   166→            )
   167→   167→
   168→   168→            // ── Status bar (when running) ──
   169→   169→            if (isRunning) {
   170→   170→                RunningStatusBar(state = state)
   171→   171→            }
   172→   172→
   173→   173→            // ── Accessibility banner ──
   174→   174→            if (!isAccessibilityEnabled) {
   175→   175→                AccessibilityBanner(onEnable = { viewModel.openAccessibilitySettings() })
   176→   176→            }
   177→   177→
   178→   178→            // ── Result banners ──
   179→   179→            if (state.status == AgentStatus.COMPLETED && !isRunning) {
   180→   180→                ResultBanner(text = "Done", isSuccess = true)
   181→   181→            } else if (state.status == AgentStatus.FAILED && !isRunning) {
   182→   182→                ResultBanner(text = state.lastError ?: "Something went wrong", isSuccess = false)
   183→   183→            } else if (state.status == AgentStatus.CANCELLED && !isRunning) {
   184→   184→                ResultBanner(text = "Stopped", isSuccess = false)
   185→   185→            }
   186→   186→
   187→   187→            // ── Chat / Event List ──
   188→   188→            if (displayItems.isEmpty() && !isRunning) {
   189→   189→                EmptyState(onSuggestionClick = { sendTask(it) })
   190→   190→            } else {
   191→   191→                LazyColumn(
   192→   192→                    state = listState,
   193→   193→                    modifier = Modifier
   194→   194→                        .fillMaxSize()
   195→   195→                        .padding(horizontal = 16.dp),
   196→   196→                    verticalArrangement = Arrangement.spacedBy(6.dp),
   197→   197→                    contentPadding = PaddingValues(vertical = 8.dp)
   198→   198→                ) {
   199→   199→                    items(
   200→   200→                        items = displayItems,
   201→   201→                        key = { it.timestamp.toString() + it::class.simpleName }
   202→   202→                    ) { item ->
   203→   203→                        when (item) {
   204→   204→                            is DisplayItem.UserMessage -> UserChatBubble(text = item.text)
   205→   205→                            is DisplayItem.AgentMessage -> AgentChatBubble(text = item.text)
   206→   206→                            is DisplayItem.ToolAction -> ToolActionCard(item)
   207→   207→                            is DisplayItem.ErrorItem -> ErrorText(item.message)
   208→   208→                        }
   209→   209→                    }
   210→   210→                    // Bottom padding for input bar
   211→   211→                    item {
   212→   212→                        Spacer(modifier = Modifier.height(80.dp))
   213→   213→                    }
   214→   214→                }
   215→   215→            }
   216→   216→        }
   217→   217→
   218→   218→        // ── Bottom Input Bar ──
   219→   219→        BottomInputBar(
   220→   220→            input = taskInput,
   221→   221→            onInputChange = { taskInput = it },
   222→   222→            onSend = { sendTask(taskInput) },
   223→   223→            onStop = { viewModel.stopAgent() },
   224→   224→            isRunning = isRunning,
   225→   225→            isListening = isListening,
   226→   226→            onVoiceClick = { startVoiceInput() },
   227→   227→            canSend = taskInput.isNotBlank() && !isRunning
   228→   228→        )
   229→   229→
   230→   230→        // ── Dialogs ──
   231→   231→        state.pendingConfirmation?.let { confirmation ->
   232→   232→            ConfirmationDialog(
   233→   233→                confirmation = confirmation,
   234→   234→                onConfirm = { viewModel.respondToConfirmation(true) },
   235→   235→                onDeny = { viewModel.respondToConfirmation(false) }
   236→   236→            )
   237→   237→        }
   238→   238→
   239→   239→        state.pendingQuestion?.let { question ->
   240→   240→            UserQuestionDialog(
   241→   241→                question = question,
   242→   242→                onAnswer = { answer -> viewModel.respondToUser(answer) },
   243→   243→                onDismiss = {}
   244→   244→            )
   245→   245→        }
   246→   246→    }
   247→   247→}

   735→   735→private fun describeAction(toolName: String, arguments: String): String {
   736→   736→    val text = extractJsonField(arguments, "text")
   737→   737→    val appName = extractJsonField(arguments, "app_name")
   738→   738→    val direction = extractJsonField(arguments, "direction")
   739→   739→    return when {
   740→   740→        toolName.contains("launch_app") -> if (!appName.isNullOrBlank()) "Opened $appName" else "Opened app"
   741→   741→        toolName.contains("click") -> "Tapped element"
   742→   742→        toolName.contains("long_click") -> "Long pressed"
   743→   743→        toolName.contains("type_text") -> if (!text.isNullOrBlank()) "Typed: ${text.take(40)}" else "Typed text"
   744→   744→        toolName.contains("clear_text") -> "Cleared text"
   745→   745→        toolName.contains("scroll") -> "Scrolled ${direction ?: ""}"
   746→   746→        toolName.contains("swipe") -> {
   747→   747→            val d = extractJsonField(arguments, "direction") ?: ""
   748→   748→            when {
   749→   749→                d.contains("up", ignoreCase = true) -> "Swiped up"
   750→   750→                d.contains("down", ignoreCase = true) -> "Swiped down"
   751→   751→                d.contains("left", ignoreCase = true) -> "Swiped left"
   752→   752→                d.contains("right", ignoreCase = true) -> "Swiped right"
   753→   753→                else -> "Swiped"
   754→   754→            }
   755→   755→        }
   756→   756→        toolName.contains("back") -> "Went back"
   757→   757→        toolName.contains("home") -> "Went home"
   758→   758→        toolName.contains("recents") -> "Opened recents"
   759→   759→        toolName.contains("press_key") -> "Pressed key"
   760→   760→        toolName.contains("wait") -> "Waiting..."
   761→   761→        toolName.contains("screenshot") -> "Took screenshot"
   762→   762→        toolName.contains("inspect") -> "Read screen"
   763→   763→        toolName.contains("find") -> "Searching..."
   764→   764→        toolName.contains("analyze") -> "Analyzing screen"
   765→   765→        toolName.contains("visual") -> "Looking for element"
   766→   766→        toolName.contains("finish") -> "Done"
   767→   767→        toolName.contains("stop") -> "Stopped"
   768→   768→        else -> toolName.substringAfterLast(".").replace("_", " ")
   769→   769→    }
   770→   770→}
