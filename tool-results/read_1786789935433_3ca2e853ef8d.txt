    45→@OptIn(ExperimentalMaterial3Api::class)
    46→@Composable
    47→fun MainScreen(
    48→    viewModel: AgentViewModel,
    49→    onNavigateToSettings: () -> Unit,
    50→    onNavigateToDebug: () -> Unit,
    51→    onNavigateToHistory: () -> Unit
    52→) {
    53→    val state by viewModel.stateFlow.collectAsState()
    54→    val isAccessibilityEnabled by remember {
    55→        mutableStateOf(viewModel.isAccessibilityServiceEnabled())
    56→    }
    57→    var taskInput by remember { mutableStateOf("") }
    58→    val listState = rememberLazyListState()
    59→    val isRunning = state.status.isActive
    60→    val context = LocalContext.current
    61→
    62→    var isListening by remember { mutableStateOf(false) }
    63→
    64→    val displayItems = remember(state.history) {
    65→        state.history.mapNotNull { event ->
    66→            when (event) {
    67→                is AgentEvent.UserMessage -> DisplayItem.UserMessage(event.text, event.timestamp)
    68→                is AgentEvent.AgentMessage -> DisplayItem.AgentMessage(event.text, event.timestamp)
    69→                is AgentEvent.ToolExecution -> DisplayItem.ToolAction(
    70→                    name = event.toolName,
    71→                    args = event.arguments,
    72→                    success = event.result.success,
    73→                    error = event.result.error?.message
    74→                )
    75→                is AgentEvent.Error -> DisplayItem.ErrorItem(event.message)
    76→                else -> null
    77→            }
    78→        }
    79→    }
    80→
    81→    LaunchedEffect(displayItems.size) {
    82→        if (displayItems.isNotEmpty()) {
    83→            listState.animateScrollToItem(displayItems.size - 1)
    84→        }
    85→    }
    86→
    87→    val speechRecognizer = remember {
    88→        try { SpeechRecognizer.createSpeechRecognizer(context) } catch (_: Exception) { null }
    89→    }
    90→    DisposableEffect(Unit) { onDispose { speechRecognizer?.destroy() } }
    91→
    92→    fun startVoiceInput() {
    93→        if (speechRecognizer == null) return
    94→        isListening = true
    95→        val intent = Intent(RecognizerIntent.ACTION_RECOGNIZE_SPEECH).apply {
    96→            putExtra(RecognizerIntent.EXTRA_LANGUAGE_MODEL, RecognizerIntent.LANGUAGE_MODEL_FREE_FORM)
    97→            putExtra(RecognizerIntent.EXTRA_LANGUAGE, "en-IN")
    98→            putExtra(RecognizerIntent.EXTRA_PROMPT, "Speak to Android-Use...")
    99→        }
   100→        try {
   101→            speechRecognizer.setRecognitionListener(object : RecognitionListener {
   102→                override fun onReadyForSpeech(p: Bundle?) {}
   103→                override fun onBeginningOfSpeech() {}
   104→                override fun onRmsChanged(r: Float) {}
   105→                override fun onBufferReceived(b: ByteArray?) {}
   106→                override fun onEndOfSpeech() { isListening = false }
   107→                override fun onError(e: Int) { isListening = false }
   108→                override fun onResults(results: Bundle?) {
   109→                    val matches = results?.getStringArrayList(SpeechRecognizer.RESULTS_RECOGNITION)
   110→                    if (!matches.isNullOrEmpty()) taskInput = matches.first()
   111→                    isListening = false
   112→                }
   113→                override fun onPartialResults(p: Bundle?) {}
   114→                override fun onEvent(e: Int, p: Bundle?) {}
   115→            })
   116→            speechRecognizer.startListening(intent)
   117→        } catch (_: ActivityNotFoundException) { isListening = false }
   118→    }
   119→
   120→    fun sendTask(text: String) {
   121→        val trimmed = text.trim()
   122→        if (trimmed.isNotBlank()) {
   123→            if (state.goal.isBlank()) viewModel.startTask(trimmed)
   124→            else viewModel.continueChat(trimmed)
   125→            taskInput = ""
   126→        }
   127→    }
   128→
   129→    Box(modifier = Modifier.fillMaxSize().background(AppColors.DarkBackground)) {
   130→        Column(modifier = Modifier.fillMaxSize().align(Alignment.TopStart)) {
   131→            TopBar(
   132→                onNewChat = { viewModel.newChat() },
   133→                onHistory = onNavigateToHistory,
   134→                onSettings = onNavigateToSettings,
   135→                onDebug = onNavigateToDebug
   136→            )
   137→            if (isRunning) RunningStatusBar(state = state)
   138→            if (!isAccessibilityEnabled) {
   139→                AccessibilityBanner(onEnable = { viewModel.openAccessibilitySettings() })
   140→            }
   141→            if (state.status == AgentStatus.COMPLETED && !isRunning) {
   142→                ResultBanner(text = "Done", isSuccess = true)
   143→            } else if (state.status == AgentStatus.FAILED && !isRunning) {
   144→                ResultBanner(text = state.lastError ?: "Something went wrong", isSuccess = false)
   145→            } else if (state.status == AgentStatus.CANCELLED && !isRunning) {
   146→                ResultBanner(text = "Stopped", isSuccess = false)
   147→            }
   148→
   149→            if (displayItems.isEmpty() && !isRunning) {
   150→                EmptyState(onSuggestionClick = { sendTask(it) })
   151→            } else {
   152→                LazyColumn(
   153→                    state = listState,
   154→                    modifier = Modifier.fillMaxSize().padding(horizontal = 14.dp),
   155→                    verticalArrangement = Arrangement.spacedBy(5.dp),
   156→                    contentPadding = PaddingValues(vertical = 6.dp)
   157→                ) {
   158→                    items(items = displayItems, key = { it.timestamp.toString() + it::class.simpleName }) { item ->
   159→                        when (item) {
   160→                            is DisplayItem.UserMessage -> UserChatBubble(text = item.text)
   161→                            is DisplayItem.AgentMessage -> AgentChatBubble(text = item.text)
   162→                            is DisplayItem.ToolAction -> ToolActionCard(item)
   163→                            is DisplayItem.ErrorItem -> ErrorText(item.message)
   164→                        }
   165→                    }
   166→                    item { Spacer(modifier = Modifier.height(90.dp)) }
   167→                }
   168→            }
   169→        }
   170→
   171→        // ── BOTTOM INPUT BAR — anchored to bottom ──
   172→        Box(
   173→            modifier = Modifier
   174→                .align(Alignment.BottomCenter)
   175→                .fillMaxWidth()
   176→                .background(AppColors.DarkBackground)
   177→        ) {
   178→            BottomInputBar(
   179→                input = taskInput,
   180→                onInputChange = { taskInput = it },
   181→                onSend = { sendTask(taskInput) },
   182→                onStop = { viewModel.stopAgent() },
   183→                isRunning = isRunning,
   184→                isListening = isListening,
   185→                onVoiceClick = { startVoiceInput() },
   186→                canSend = taskInput.isNotBlank() && !isRunning
   187→            )
   188→        }
   189→
   190→        state.pendingConfirmation?.let { conf ->
   191→            ConfirmationDialog(conf, onConfirm = { viewModel.respondToConfirmation(true) }, onDeny = { viewModel.respondToConfirmation(false) })
   192→        }
   193→        state.pendingQuestion?.let { q ->
   194→            UserQuestionDialog(q, onAnswer = { viewModel.respondToUser(it) }, onDismiss = {})
   195→        }
   196→    }
   197→}

   259→@Composable
   260→private fun RunningStatusBar(state: com.androidagent.aiagent.agent.AgentState) {
   261→    val (label, icon, color) = when (state.status) {
   262→        AgentStatus.THINKING -> Triple("Analyzing screen", Icons.Default.Psychology, AppColors.TextSecondary)
   263→        AgentStatus.EXECUTING -> Triple("Using mobile...", Icons.Default.TouchApp, AppColors.Success)
   264→        AgentStatus.WAITING_FOR_USER -> Triple("Waiting for you", Icons.Default.QuestionAnswer, AppColors.Primary)
   265→        AgentStatus.WAITING_FOR_CONFIRMATION -> Triple("Confirm action", Icons.Default.Security, AppColors.Warning)
   266→        AgentStatus.VERIFYING -> Triple("Verifying...", Icons.Default.VerifiedUser, AppColors.TextSecondary)
   267→        else -> Triple("Working...", Icons.Default.Autorenew, AppColors.Secondary)
   268→    }
   269→    val pulse = rememberInfiniteTransition(label = "p").animateFloat(
   270→        0.3f, 1f, infiniteRepeatable(tween(800, easing = LinearEasing), RepeatMode.Reverse), "a"
   271→    )
   272→    Row(
   273→        modifier = Modifier.fillMaxWidth().background(color.copy(alpha = 0.06f)).padding(horizontal = 14.dp, vertical = 6.dp),
   274→        verticalAlignment = Alignment.CenterVertically
   275→    ) {
   276→        Icon(icon, null, tint = color, modifier = Modifier.size(14.dp).alpha(pulse.value))
   277→        Spacer(Modifier.width(8.dp))
   278→        Text(label, color = color, fontSize = 12.5.sp, fontWeight = FontWeight.Medium, modifier = Modifier.weight(1f))
   279→        Text("${state.durationFormatted}", color = AppColors.TextMuted, fontSize = 11.sp)
   280→        Spacer(Modifier.width(10.dp))
   281→        Text("Step ${state.stepNumber}", color = AppColors.TextMuted, fontSize = 11.sp)
   282→    }
   283→}

   356→@Composable
   357→private fun BottomInputBar(
   358→    input: String, onInputChange: (String) -> Unit, onSend: () -> Unit,
   359→    onStop: () -> Unit, isRunning: Boolean, isListening: Boolean,
   360→    onVoiceClick: () -> Unit, canSend: Boolean
   361→) {

   363→        modifier = Modifier.fillMaxWidth().background(AppColors.Surface).navigationBarsPadding().padding(horizontal = 12.dp, vertical = 8.dp)
   364→    ) {
   365→        if (isListening) {
   366→            val s = rememberInfiniteTransition(label = "m").animateFloat(0.8f, 1.2f, infiniteRepeatable(tween(400), RepeatMode.Reverse), "s")
   367→            Row(modifier = Modifier.fillMaxWidth().padding(bottom = 5.dp), horizontalArrangement = Arrangement.Center) {
   368→                Icon(Icons.Default.Mic, "Listening", tint = AppColors.Error, modifier = Modifier.size(20.dp * s.value))
   369→                Spacer(Modifier.width(8.dp))
   370→                Text("Listening...", color = AppColors.Error, fontSize = 13.sp, fontWeight = FontWeight.Medium)
   371→            }
   372→        }
   373→        Surface(
   374→            color = AppColors.SurfaceVariant, shape = RoundedCornerShape(24.dp),
   375→            border = BorderStroke(1.dp, AppColors.Line), modifier = Modifier.fillMaxWidth()
   376→        ) {
   377→            Row(modifier = Modifier.padding(horizontal = 4.dp, vertical = 2.dp), verticalAlignment = Alignment.CenterVertically) {
   378→                IconButton(onClick = onVoiceClick, modifier = Modifier.size(42.dp)) {
   379→                    Icon(
   380→                        if (isListening) Icons.Default.Mic else Icons.Default.MicNone,
   381→                        "Voice", tint = if (isListening) AppColors.Error else AppColors.TextMuted, modifier = Modifier.size(20.dp)
   382→                    )
   383→                }
   384→                BasicTextField(
   385→                    value = input, onValueChange = onInputChange, modifier = Modifier.weight(1f),
   386→                    textStyle = TextStyle(color = AppColors.TextPrimary, fontSize = 15.sp, lineHeight = 20.sp),
   387→                    cursorBrush = SolidColor(AppColors.TextPrimary),
   388→                    decorationBox = { innerTextField ->
   389→                        Box(Modifier.padding(vertical = 10.dp)) {
   390→                            if (input.isEmpty()) Text("Ask Android-Use anything...", color = AppColors.TextMuted, fontSize = 15.sp)
   391→                            innerTextField()
   392→                        }
   393→                    }
   394→                )
   395→                Box(
   396→                    modifier = Modifier.size(38.dp).clip(CircleShape).background(
   397→                        if (isRunning) AppColors.Error else if (canSend) AppColors.TextPrimary else AppColors.SurfaceHover
   398→                    ).clickable(enabled = canSend || isRunning) {
   399→                        if (isRunning) onStop() else onSend()
   400→                    }, contentAlignment = Alignment.Center
   401→                ) {
   402→                    Icon(
   403→                        if (isRunning) Icons.Default.Stop else Icons.AutoMirrored.Filled.Send,
   404→                        if (isRunning) "Stop" else "Send",
   405→                        tint = when { isRunning -> Color.White; canSend -> AppColors.DarkBackground; else -> AppColors.TextMuted },
   406→                        modifier = Modifier.size(18.dp)
   407→                    )
   408→                }
   409→            }
   410→        }
   411→    }
   412→}

   577→private fun extractJsonField(json: String, field: String): String? {
   578→    return Regex("""$field"\s*:\s*"([^"\x27]*)"""", RegexOption.IGNORE_CASE).find(json)?.groupValues?.getOrNull(1)
