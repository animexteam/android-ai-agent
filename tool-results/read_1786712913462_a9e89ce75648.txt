    43→@OptIn(ExperimentalMaterial3Api::class)
    44→@Composable
    45→fun MainScreen(
    46→    viewModel: AgentViewModel,
    47→    onNavigateToSettings: () -> Unit,
    48→    onNavigateToDebug: () -> Unit,
    49→    onNavigateToHistory: () -> Unit
    50→) {
    51→    val state by viewModel.stateFlow.collectAsState()
    52→    val isAccessibilityEnabled by remember {
    53→        mutableStateOf(viewModel.isAccessibilityServiceEnabled())
    54→    }
    55→    var taskInput by remember { mutableStateOf("") }
    56→    val listState = rememberLazyListState()
    57→    val isRunning = state.status.isActive
    58→    val context = LocalContext.current
    59→
    60→    // Voice input state
    61→    var isListening by remember { mutableStateOf(false) }
    62→    var voiceText by remember { mutableStateOf("") }
    63→
    64→    // Build display items: chat messages + tool actions
    65→    val displayItems = remember(state.history) {
    66→        state.history.mapNotNull { event ->
    67→            when (event) {
    68→                is AgentEvent.UserMessage -> DisplayItem.UserMessage(event.text, event.timestamp)
    69→                is AgentEvent.AgentMessage -> DisplayItem.AgentMessage(event.text, event.timestamp)
    70→                is AgentEvent.ToolExecution -> DisplayItem.ToolAction(
    71→                    name = event.toolName,
    72→                    args = event.arguments,
    73→                    success = event.result.success,
    74→                    error = event.result.error?.message
    75→                )
    76→                is AgentEvent.Error -> DisplayItem.ErrorItem(event.message)
    77→                else -> null
    78→            }
    79→        }
    80→    }
    81→
    82→    LaunchedEffect(displayItems.size) {
    83→        if (displayItems.isNotEmpty()) {
    84→            listState.animateScrollToItem(displayItems.size - 1)
    85→        }
    86→    }
    87→
    88→    // Voice recognizer
    89→    val speechRecognizer = remember {
    90→        try {
    91→            SpeechRecognizer.createSpeechRecognizer(context)
    92→        } catch (e: Exception) { null }
    93→    }
    94→
    95→    DisposableEffect(Unit) {
    96→        onDispose {
    97→            speechRecognizer?.destroy()
    98→        }
    99→    }
   100→
   101→    fun startVoiceInput() {
   102→        if (speechRecognizer == null) return
   103→        isListening = true
   104→        voiceText = ""
   105→        val intent = Intent(RecognizerIntent.ACTION_RECOGNIZE_SPEECH).apply {
   106→            putExtra(RecognizerIntent.EXTRA_LANGUAGE_MODEL, RecognizerIntent.LANGUAGE_MODEL_FREE_FORM)
   107→            putExtra(RecognizerIntent.EXTRA_LANGUAGE, "en-IN")
   108→            putExtra(RecognizerIntent.EXTRA_PROMPT, "Speak to Android-Use...")
   109→        }
   110→        try {
   111→            speechRecognizer.setRecognitionListener(object : RecognitionListener {
   112→                override fun onReadyForSpeech(params: Bundle?) {}
   113→                override fun onBeginningOfSpeech() {}
   114→                override fun onRmsChanged(rmsdB: Float) {}
   115→                override fun onBufferReceived(buffer: ByteArray?) {}
   116→                override fun onEndOfSpeech() {
   117→                    isListening = false
   118→                }
   119→                override fun onError(error: Int) {
   120→                    isListening = false
   121→                }
   122→                override fun onResults(results: Bundle?) {
   123→                    val matches = results?.getStringArrayList(SpeechRecognizer.RESULTS_RECOGNITION)
   124→                    val text = matches?.firstOrNull() ?: ""
   125→                    if (text.isNotBlank()) {
   126→                        taskInput = text
   127→                    }
   128→                    isListening = false
   129→                }
   130→                override fun onPartialResults(partialResults: Bundle?) {
   131→                    val matches = partialResults?.getStringArrayList(SpeechRecognizer.RESULTS_RECOGNITION)
   132→                    voiceText = matches?.firstOrNull() ?: ""
   133→                }
   134→                override fun onEvent(eventType: Int, params: Bundle?) {}
   135→            })
   136→            speechRecognizer.startListening(intent)
   137→        } catch (e: ActivityNotFoundException) {
   138→            isListening = false
   139→        }
   140→    }
   141→
   142→    Box(
   143→        modifier = Modifier
   144→            .fillMaxSize()
   145→            .background(AppColors.DarkBackground)
   146→    ) {
   147→        Column(modifier = Modifier.fillMaxSize()) {
   148→            // ── Top Bar ──
   149→            TopBar(
   150→                onNewChat = { viewModel.newChat() },
   151→                onHistory = onNavigateToHistory,
   152→                onSettings = onNavigateToSettings
   153→            )
   154→
   155→            // ── Status bar (when running) ──
   156→            if (isRunning) {
   157→                RunningStatusBar(state = state)
   158→            }
   159→
   160→            // ── Accessibility banner ──
   161→            if (!isAccessibilityEnabled) {
   162→                AccessibilityBanner(onEnable = { viewModel.openAccessibilitySettings() })
   163→            }
   164→
   165→            // ── Result banners ──
   166→            if (state.status == AgentStatus.COMPLETED && !isRunning) {
   167→                ResultBanner(text = "Done", isSuccess = true)
   168→            } else if (state.status == AgentStatus.FAILED && !isRunning) {
   169→                ResultBanner(text = state.lastError ?: "Something went wrong", isSuccess = false)
   170→            } else if (state.status == AgentStatus.CANCELLED && !isRunning) {
   171→                ResultBanner(text = "Stopped", isSuccess = false)
   172→            }
   173→
   174→            // ── Chat / Event List ──
   175→            if (displayItems.isEmpty() && !isRunning) {
   176→                EmptyState()
   177→            } else {
   178→                LazyColumn(
   179→                    state = listState,
   180→                    modifier = Modifier
   181→                        .fillMaxSize()
   182→                        .padding(horizontal = 16.dp),
   183→                    verticalArrangement = Arrangement.spacedBy(6.dp),
   184→                    contentPadding = PaddingValues(vertical = 8.dp)
   185→                ) {
   186→                    items(
   187→                        items = displayItems,
   188→                        key = { it.timestamp.toString() + it::class.simpleName }
   189→                    ) { item ->
   190→                        when (item) {
   191→                            is DisplayItem.UserMessage -> UserChatBubble(text = item.text)
   192→                            is DisplayItem.AgentMessage -> AgentChatBubble(text = item.text)
   193→                            is DisplayItem.ToolAction -> ToolActionCard(item)
   194→                            is DisplayItem.ErrorItem -> ErrorText(item.message)
   195→                        }
   196→                    }
   197→                    // Bottom padding for input bar
   198→                    item {
   199→                        Spacer(modifier = Modifier.height(80.dp))
   200→                    }
   201→                }
   202→            }
   203→        }
   204→
   205→        // ── Bottom Input Bar ──
   206→        BottomInputBar(
   207→            input = taskInput,
   208→            onInputChange = { taskInput = it },
   209→            onSend = {
   210→                val trimmed = taskInput.trim()
   211→                if (trimmed.isNotBlank()) {
   212→                    if (state.goal.isBlank()) {
   213→                        viewModel.startTask(trimmed)
   214→                    } else {
   215→                        viewModel.continueChat(trimmed)
   216→                    }
   217→                    taskInput = ""
   218→                }
   219→            },
   220→            onStop = { viewModel.stopAgent() },
   221→            isRunning = isRunning,
   222→            isListening = isListening,
   223→            onVoiceClick = { startVoiceInput() },
   224→            canSend = taskInput.isNotBlank() && !isRunning
   225→        )
   226→
   227→        // ── Dialogs ──
   228→        state.pendingConfirmation?.let { confirmation ->
   229→            ConfirmationDialog(
   230→                confirmation = confirmation,
   231→                onConfirm = { viewModel.respondToConfirmation(true) },
   232→                onDeny = { viewModel.respondToConfirmation(false) }
   233→            )
   234→        }
   235→
   236→        state.pendingQuestion?.let { question ->
   237→            UserQuestionDialog(
   238→                question = question,
   239→                onAnswer = { answer -> viewModel.respondToUser(answer) },
   240→                onDismiss = {}
   241→            )
   242→        }
   243→    }
   244→}

   330→@Composable
   331→private fun RunningStatusBar(state: com.androidagent.aiagent.agent.AgentState) {
   332→    val (label, icon, color) = when (state.status) {
   333→        AgentStatus.THINKING -> Triple("Analyzing screen", Icons.Default.Psychology, AppColors.Info)
   334→        AgentStatus.EXECUTING -> Triple("Using mobile...", Icons.Default.TouchApp, AppColors.Success)
   335→        AgentStatus.WAITING_FOR_USER -> Triple("Waiting for you", Icons.Default.QuestionAnswer, AppColors.Primary)
   336→        AgentStatus.WAITING_FOR_CONFIRMATION -> Triple("Confirm action", Icons.Default.Security, AppColors.Warning)
   337→        AgentStatus.VERIFYING -> Triple("Verifying...", Icons.Default.VerifiedUser, AppColors.Info)
   338→        else -> Triple("Working...", Icons.Default.Autorenew, AppColors.Secondary)
   339→    }
   340→
   341→    val infiniteTransition = rememberInfiniteTransition(label = "pulse")
   342→    val alpha by infiniteTransition.animateFloat(
   343→        initialValue = 0.3f, targetValue = 1f,
   344→        animationSpec = infiniteRepeatable(tween(800, easing = LinearEasing), RepeatMode.Reverse),
   345→        label = "alpha"
   346→    )
   347→
   348→    Row(
   349→        modifier = Modifier
   350→            .fillMaxWidth()
   351→            .background(color.copy(alpha = 0.08f))
   352→            .padding(horizontal = 16.dp, vertical = 8.dp),
   353→        verticalAlignment = Alignment.CenterVertically
   354→    ) {
   355→        Icon(icon, contentDescription = null, tint = color, modifier = Modifier.size(16.dp).alpha(alpha))
   356→        Spacer(modifier = Modifier.width(10.dp))
   357→        Text(label, color = color, fontSize = 13.sp, fontWeight = FontWeight.Medium, modifier = Modifier.weight(1f))
   358→        Text("${state.durationFormatted}", color = AppColors.TextMuted, fontSize = 11.sp)
   359→        Spacer(modifier = Modifier.width(12.dp))
   360→        Text("Step ${state.stepNumber}", color = AppColors.TextMuted, fontSize = 11.sp)
   361→    }
   362→}

   432→        Spacer(modifier = Modifier.width(8.dp))

   435→            color = color,
   436→            fontSize = 12.sp,
   437→            maxLines = 1,
   438→            modifier = Modifier.weight(1f)
   439→        )

   441→            Icon(Icons.Default.Check, contentDescription = null, tint = AppColors.Success, modifier = Modifier.size(14.dp))

   699→private fun describeAction(toolName: String, arguments: String): String {
   700→    val text = extractJsonField(arguments, "text")
   701→    val appName = extractJsonField(arguments, "app_name")
   702→    val direction = extractJsonField(arguments, "direction")
   703→    return when {
   704→        toolName.contains("launch_app") -> if (!appName.isNullOrBlank()) "Opened $appName" else "Opened app"
   705→        toolName.contains("click") -> "Tapped element"
   706→        toolName.contains("long_click") -> "Long pressed"
   707→        toolName.contains("type_text") -> if (!text.isNullOrBlank()) "Typed: ${text.take(40)}" else "Typed text"
   708→        toolName.contains("clear_text") -> "Cleared text"
   709→        toolName.contains("scroll") -> "Scrolled ${direction ?: ""}"
   710→        toolName.contains("swipe") -> {
   711→            val d = extractJsonField(arguments, "direction") ?: ""
   712→            when {
   713→                d.contains("up", ignoreCase = true) -> "Swiped up"
   714→                d.contains("down", ignoreCase = true) -> "Swiped down"
   715→                d.contains("left", ignoreCase = true) -> "Swiped left"
   716→                d.contains("right", ignoreCase = true) -> "Swiped right"
   717→                else -> "Swiped"
   718→            }
   719→        }
   720→        toolName.contains("back") -> "Went back"
   721→        toolName.contains("home") -> "Went home"
   722→        toolName.contains("recents") -> "Opened recents"
   723→        toolName.contains("press_key") -> "Pressed key"
   724→        toolName.contains("wait") -> "Waiting..."
   725→        toolName.contains("screenshot") -> "Took screenshot"
   726→        toolName.contains("inspect") -> "Read screen"
   727→        toolName.contains("find") -> "Searching..."
   728→        toolName.contains("analyze") -> "Analyzing screen"
   729→        toolName.contains("visual") -> "Looking for element"
   730→        toolName.contains("finish") -> "Done"
   731→        toolName.contains("stop") -> "Stopped"
   732→        else -> toolName.substringAfterLast(".").replace("_", " ")
   733→    }
   734→}

   753→private fun extractJsonField(json: String, field: String): String? {
   754→    return Regex(""""$field"\s*:\s*"([^"']*)"""", RegexOption.IGNORE_CASE).find(json)?.groupValues?.getOrNull(1)
   755→}
